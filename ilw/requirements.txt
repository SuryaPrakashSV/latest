WITH latest_tasks AS (
    SELECT
        "key" AS task_id,
        "status" AS general_status,
        "customStatusId" AS detailed_status,
        TRY_TO_TIMESTAMP_TZ(
            "updatedDate"::VARCHAR
        ) AS task_updated_at
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
    WHERE "id" = 'MQAAAAEKLDyQ'
      AND NULLIF(TRIM("key"), '') IS NOT NULL
      AND UPPER(TRIM("key")) <> 'PLACEHOLDER'
      AND "key" <> "id"
      AND "key" IN (
          "parent_task_id",
          "child_task_id",
          "grandchild_task_id",
          "baby_task_id",
          "grandbaby_task_id",
          "great_grandbaby_task_id"
      )
    QUALIFY ROW_NUMBER() OVER (
        PARTITION BY "key"
        ORDER BY task_updated_at DESC NULLS LAST,
                 "status",
                 "customStatusId"
    ) = 1
)
SELECT
    general_status,
    detailed_status,
    COUNT(*) AS task_count,
    SUM(IFF(
        task_updated_at >= DATEADD('hour', -48, CURRENT_TIMESTAMP()),
        1, 0
    )) AS tasks_updated_last_48h,
    MAX(task_updated_at) AS latest_task_update
FROM latest_tasks
GROUP BY general_status, detailed_status
ORDER BY task_count DESC, detailed_status;
