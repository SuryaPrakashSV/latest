SELECT
    task_id,
    COUNT(*) AS row_count,
    COUNT(DISTINCT "customStatusId") AS distinct_statuses,
    ARRAY_AGG(DISTINCT "status") AS general_statuses_seen,
    ARRAY_AGG(DISTINCT "customStatusId") AS custom_statuses_seen
FROM (
    SELECT
        COALESCE(
            "great_grandbaby_task_id",
            "grandbaby_task_id",
            "baby_task_id",
            "grandchild_task_id",
            "child_task_id",
            "parent_task_id"
        ) AS task_id,
        "status",
        "customStatusId"
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
    WHERE "id" = 'MQAAAABnIKFK'
)
WHERE task_id IS NOT NULL
GROUP BY task_id
ORDER BY row_count DESC;
