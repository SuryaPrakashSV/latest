"""Review saved Akash comparison inputs and logs. No API calls or token needed.

Save beside validate_local_baseline.py in Desktop/wrike-local-baseline.
Run: python3 review_akash_comparison.py
If there are several comparison folders, use --comparison PATH explicitly.
All source files stay unchanged. Outputs are candidates, not access verdicts.
"""
import argparse
import csv
import hashlib
import json
import re
import sys
from datetime import datetime, timezone
from pathlib import Path

import validate_local_baseline as baseline


EXTRA_CONTAINERS = {
    "IEADQP2VI5CGZ6N7": "Intake Form - Resourcing Template",
    "IEADQP2VI5CGZFX4": "NPI Project Blueprints",
    "IEADQP2VI5IWLCDD": "Test Import",
    "IEADQP2VI5DX7CRB": "xxOLD Blueprints",
}


def select_comparison(root, requested):
    if requested:
        path = requested.expanduser().resolve()
        if not (path / "summary.json").is_file():
            raise ValueError("No summary.json in comparison folder: " + str(path))
        return path
    matches = sorted((root / "output" / "akash_comparison").glob("*/summary.json"))
    if len(matches) != 1:
        options = "\n".join(str(p.parent) for p in matches)
        raise ValueError("Expected one Akash comparison folder; found {}. Use --comparison PATH.\n{}".format(len(matches), options))
    return matches[0].parent


def review_logs(root, export_path, row_count):
    findings = []
    for path in sorted((root / "output").glob("akash_run_*.log")):
        before = baseline.fingerprint(path)
        raw = path.read_bytes()
        if baseline.fingerprint(path) != before:
            raise ValueError("Log changed while reading: " + str(path))
        lines = raw.decode("utf-8", errors="replace").splitlines()
        if not any(line.strip() == "Saved: " + str(export_path) for line in lines):
            continue
        patterns = {
            "rate_limit_messages": r"\[429\].*rate limited",
            "request_retry_messages": r"Request error for",
            "worker_errors": r"^Error fetching ",
            "target_discovery_errors": r"Could not auto-fetch target folders",
            "tracebacks": r"Traceback \(most recent call last\)",
        }
        line_numbers = {
            label: [n for n, line in enumerate(lines, 1) if re.search(pattern, line, re.I)]
            for label, pattern in patterns.items()
        }
        worker_ids = sorted(set(
            match.group(1) for line in lines
            for match in [re.match(r"Error fetching ([A-Za-z0-9_-]+):", line)] if match
        ))
        findings.append({
            "path": str(path.resolve()),
            "sha256": hashlib.sha256(raw).hexdigest(),
            "saved_marker_count": sum(line.strip().startswith("Saved: ") for line in lines),
            "counts_apply_to": "Whole log file; do not attribute to one run if multiple Saved markers exist.",
            "export_row_count_message_matches": any(
                line.strip() == "Dataset rows exported: {:,}".format(row_count) for line in lines
            ),
            "message_counts": {label: len(numbers) for label, numbers in line_numbers.items()},
            "message_line_numbers": line_numbers,
            "worker_error_ids": worker_ids,
        })
    status = "NO_MATCHING_LOG" if not findings else "MULTIPLE_MATCHING_LOGS" if len(findings) > 1 else "ONE_MATCHING_LOG"
    if any(item["saved_marker_count"] != 1 for item in findings):
        status = "AMBIGUOUS_LOG_CONTENT"
    return {
        "status": status,
        "logs": findings,
        "interpretation": "Log indicators only. Retries may recover; zero logged failures does not prove API completeness. The extraction can omit empty results or fall through after a final 429.",
    }


def save_csv(path, columns, rows):
    with path.open("x", encoding="utf-8-sig", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=columns)
        writer.writeheader()
        writer.writerows(rows)


def run(root, folder):
    previous = json.loads((folder / "summary.json").read_text(encoding="utf-8-sig"))
    sources = previous["inputs"]
    sf, sf_audit = baseline.load_inventory(Path(sources["snowflake"]["path"]))
    ref, ref_audit = baseline.load_inventory(Path(sources["akash_reference"]["path"]), reference=True)
    local_path = Path(sources["local"]["path"])
    akash, akash_audit = baseline.load_local(local_path)
    audits = {"snowflake": sf_audit, "akash_reference": ref_audit, "local": akash_audit}
    for source, audit in audits.items():
        if audit["sha256"] != sources[source]["sha256"]:
            raise ValueError("Input differs from the saved comparison: " + source)
    differences = baseline.compare(sf, akash)
    counts = {name: len(ids) for name, ids in differences.items()}
    if counts != previous["counts"]:
        raise ValueError("Recomputed ID/metadata counts differ from the saved comparison.")

    missing = differences["snowflake_only"]
    new = differences["local_only"]
    ref_ids = set(ref)
    candidate_rows = []
    for entity_id in sorted(missing):
        item = sf[entity_id]
        candidate_rows.append({
            "project_id": entity_id,
            "snowflake_name": item["name"],
            "is_confirmed_project_in_snowflake": item["project"],
            "in_akash_reference": int(entity_id in ref),
            "reference_name": ref.get(entity_id, {}).get("name", ""),
            "status": "ABSENT_FROM_AKASH_EXPORT_NOT_YET_ACCESS_VERIFIED",
        })
    reference_rows = []
    for entity_id in sorted(ref):
        reference_rows.append({
            "project_id": entity_id,
            "reference_name": ref[entity_id]["name"],
            "in_snowflake_baseline": int(entity_id in sf),
            "in_akash_export": int(entity_id in akash),
            "in_missing_candidates": int(entity_id in missing),
            "snowflake_name": sf.get(entity_id, {}).get("name", ""),
            "akash_name": akash.get(entity_id, {}).get("name", ""),
        })
    local_only_rows = [{
        "project_id": entity_id,
        "akash_name": akash[entity_id]["name"],
        "is_confirmed_project_in_akash_export": akash[entity_id]["project"],
        "in_akash_reference": int(entity_id in ref),
    } for entity_id in sorted(new)]
    reference_counts = {
        "reference_total": len(ref),
        "missing_candidates_in_reference": len(missing & ref_ids),
        "missing_candidates_outside_reference": len(missing - ref_ids),
        "reference_ids_present_in_akash_export": len(ref_ids & set(akash)),
        "reference_ids_outside_baseline_and_absent_from_akash": len(ref_ids - set(sf) - set(akash)),
    }
    additional = [{
        "project_id": entity_id, "previously_reported_name": name,
        "in_baseline": entity_id in sf, "in_akash_export": entity_id in akash,
        "in_missing_candidates": entity_id in missing,
    } for entity_id, name in EXTRA_CONTAINERS.items()]
    logs = review_logs(root, local_path.resolve(), akash_audit["data_rows"])
    report = {
        "comparison_folder": str(folder.resolve()), "verified_inputs": audits,
        "counts": counts, "reference_crosscheck": reference_counts,
        "missing_confirmed_projects_in_baseline": sum(sf[i]["project"] for i in missing),
        "missing_not_confirmed_as_projects_in_baseline": sum(not sf[i]["project"] for i in missing),
        "missing_with_blank_baseline_names": sum(not sf[i]["name"] for i in missing),
        "distinct_ids_with_metadata_differences": len(differences["name_changes"] | differences["classification_changes"]),
        "previous_four_containers": additional, "log_review": logs,
        "interpretation": "Present in the saved Snowflake baseline but absent from Akash's completed export. These are candidates for targeted checking, not proven access failures. Reference membership is not an access verdict. Akash-only IDs may reflect scope or snapshot changes.",
    }
    run_id = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%S_%fZ")
    destination = folder / ("access_review_" + run_id)
    destination.mkdir(exist_ok=False)
    save_csv(destination / "missing_candidates.csv", [
        "project_id", "snowflake_name", "is_confirmed_project_in_snowflake",
        "in_akash_reference", "reference_name", "status"
    ], candidate_rows)
    save_csv(destination / "reference_crosscheck.csv", [
        "project_id", "reference_name", "in_snowflake_baseline", "in_akash_export",
        "in_missing_candidates", "snowflake_name", "akash_name"
    ], reference_rows)
    save_csv(destination / "akash_only.csv", [
        "project_id", "akash_name", "is_confirmed_project_in_akash_export", "in_akash_reference"
    ], local_only_rows)
    with (destination / "review_summary.json").open("x", encoding="utf-8") as handle:
        json.dump(report, handle, indent=2, ensure_ascii=False)

    print("Saved input hashes and comparison counts verified.")
    print("Baseline IDs: {:,}; Akash export IDs: {:,}; shared: {:,}".format(len(sf), len(akash), counts["shared"]))
    print("Missing candidates: {}; Akash-only IDs: {}".format(len(missing), len(new)))
    print("Missing confirmed projects: {}; not confirmed: {}; blank baseline names: {}".format(
        report["missing_confirmed_projects_in_baseline"], report["missing_not_confirmed_as_projects_in_baseline"], report["missing_with_blank_baseline_names"]))
    for label, number in reference_counts.items():
        print(label.replace("_", " ").capitalize() + ": " + str(number))
    print("Previous four containers:")
    for item in additional:
        state = "MISSING CANDIDATE" if item["in_missing_candidates"] else "PRESENT IN AKASH EXPORT" if item["in_akash_export"] else "OUTSIDE BASELINE"
        print("  " + item["previously_reported_name"] + ": " + state)
    print("Log review: " + logs["status"])
    for log in logs["logs"]:
        print("  Log: " + log["path"])
        print("  Saved markers: " + str(log["saved_marker_count"]) + "; counts cover the whole log file")
        print("  Export row-count message matches: " + str(log["export_row_count_message_matches"]))
        for label, number in log["message_counts"].items():
            print("  " + label.replace("_", " ") + ": " + str(number))
    print("Log counts do not establish API completeness or access permissions.")
    print("Saved review: " + str(destination.resolve()))
    print("Local-file review complete. No API calls were made; no source files were modified.")
    return report, destination


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--comparison", type=Path, help="Exact folder containing the Akash comparison summary.json")
    args = parser.parse_args()
    root = Path(__file__).resolve().parent
    run(root, select_comparison(root, args.comparison))


if __name__ == "__main__":
    try:
        main()
    except (OSError, ValueError, KeyError, TypeError, csv.Error) as exc:
        print("Review error: " + str(exc), file=sys.stderr)
        sys.exit(1)
