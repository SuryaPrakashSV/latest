WITH task_rows AS (
    SELECT
        ROW_NUMBER() OVER (
            ORDER BY updatedDate, completedDate
        ) AS physical_row_number,
        w.*
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG" w
    WHERE parent_task_id = 'IEADQP2VKRD2ARI6'
       OR child_task_id = 'IEADQP2VKRD2ARI6'
       OR grandchild_task_id = 'IEADQP2VKRD2ARI6'
       OR baby_task_id = 'IEADQP2VKRD2ARI6'
       OR grandbaby_task_id = 'IEADQP2VKRD2ARI6'
       OR great_grandbaby_task_id = 'IEADQP2VKRD2ARI6'
),
task_objects AS (
    SELECT
        physical_row_number,
        OBJECT_CONSTRUCT_KEEP_NULL(task_rows.*) AS task_record
    FROM task_rows
)
SELECT
    physical_row_number,
    f.key::STRING AS field_name,
    f.value AS field_value
FROM task_objects,
LATERAL FLATTEN(INPUT => task_record) f
WHERE REGEXP_LIKE(
    f.key::STRING,
    'status|date|time|start|end|created|updated|completed',
    'i'
)
ORDER BY physical_row_number, field_name;

SELECT
    status,
    customStatusId,
    COUNT(*) AS physical_row_count
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
GROUP BY status, customStatusId
ORDER BY status, customStatusId;
