SELECT
    "key" AS task_id,
    "id" AS project_id,
    "title" AS task_title,
    "status" AS general_status,
    "customStatusId" AS detailed_status,
    "updatedDate" AS updated_date,
    "completedDate" AS completed_date,
    COUNT(*) AS physical_row_count
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
WHERE "key" = 'IEADQP2VKRD2ARI6'
GROUP BY
    "key",
    "id",
    "title",
    "status",
    "customStatusId",
    "updatedDate",
    "completedDate"
ORDER BY
    updated_date,
    completed_date;
