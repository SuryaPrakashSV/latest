"""Table-wide search: do any of the 123 tasks 'missing from Snowflake' for
Holiday actually appear ANYWHERE ELSE in Wrike_STG - under some different
project entirely?

Connects directly to Snowflake (reusing the same SNOWFLAKE_* env-var pattern
already used elsewhere in this project) and loads the local CSV of missing
IDs into a temporary table, then joins it against a full-table (no project
filter) union of every hierarchy column. This avoids retyping any of the
123 IDs by hand - they go straight from the CSV file into Snowflake.

Requires the same SNOWFLAKE_* environment variables already used for
ensure-table/run: SNOWFLAKE_ACCOUNT, SNOWFLAKE_USER, SNOWFLAKE_WAREHOUSE,
SNOWFLAKE_ROLE, and SNOWFLAKE_AUTHENTICATOR=externalbrowser for SSO.

Usage:
    python3 table_wide_search.py \
        --missing-csv output/task_reconciliation/MQAAAAEIsSOV_api_only_missing_from_snowflake.csv
"""

from __future__ import annotations

import argparse
import csv
import os

TABLE_FQN = 'GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"'


def read_task_ids(csv_path: str) -> list[str]:
    with open(csv_path, newline="", encoding="utf-8-sig") as f:
        reader = csv.DictReader(f)
        fieldnames = {(name or "").strip().lower(): name for name in (reader.fieldnames or [])}
        key = fieldnames.get("task_id")
        if key is None:
            raise SystemExit(f"No task_id column found in {csv_path}. Columns: {reader.fieldnames}")
        return [row[key].strip() for row in reader if row.get(key, "").strip()]


def snowflake_connection_from_env():
    import snowflake.connector

    authenticator = os.getenv("SNOWFLAKE_AUTHENTICATOR", "snowflake").strip().lower()
    base_required = ["SNOWFLAKE_ACCOUNT", "SNOWFLAKE_USER", "SNOWFLAKE_WAREHOUSE"]
    required = base_required if authenticator == "externalbrowser" else base_required + ["SNOWFLAKE_PASSWORD"]
    missing = [name for name in required if not os.getenv(name)]
    if missing:
        raise SystemExit(
            f"Missing required environment variable(s) for Snowflake: {missing}. "
            f"(authenticator={authenticator!r} - set SNOWFLAKE_AUTHENTICATOR=externalbrowser "
            "for SSO/AppleConnect login, same as before.)"
        )

    kwargs = dict(
        account=os.environ["SNOWFLAKE_ACCOUNT"],
        user=os.environ["SNOWFLAKE_USER"],
        warehouse=os.environ["SNOWFLAKE_WAREHOUSE"],
        role=os.getenv("SNOWFLAKE_ROLE"),
    )
    if authenticator == "externalbrowser":
        kwargs["authenticator"] = "externalbrowser"
        print("Opening a browser window to complete SSO login - finish logging in there, then return here.")
    else:
        kwargs["password"] = os.environ["SNOWFLAKE_PASSWORD"]

    return snowflake.connector.connect(**kwargs)


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--missing-csv", required=True)
    args = parser.parse_args()

    missing_ids = read_task_ids(args.missing_csv)
    print(f"Loaded {len(missing_ids)} missing task IDs from {args.missing_csv}")

    connection = snowflake_connection_from_env()
    try:
        with connection.cursor() as cursor:
            cursor.execute(
                "CREATE OR REPLACE TEMPORARY TABLE TEMP_MISSING_HOLIDAY_TASKS (task_id VARCHAR)"
            )
            cursor.executemany(
                "INSERT INTO TEMP_MISSING_HOLIDAY_TASKS (task_id) VALUES (%(task_id)s)",
                [{"task_id": task_id} for task_id in missing_ids],
            )
            print(f"Loaded {len(missing_ids)} IDs into a session-scoped temporary Snowflake table.")

            query = f"""
                WITH all_hierarchy AS (
                    SELECT "id" AS owning_id, "title" AS owning_title, NULLIF(TRIM("parent_task_id"), '') AS task_id
                    FROM {TABLE_FQN}
                    UNION ALL
                    SELECT "id", "title", NULLIF(TRIM("child_task_id"), '') FROM {TABLE_FQN}
                    UNION ALL
                    SELECT "id", "title", NULLIF(TRIM("grandchild_task_id"), '') FROM {TABLE_FQN}
                    UNION ALL
                    SELECT "id", "title", NULLIF(TRIM("baby_task_id"), '') FROM {TABLE_FQN}
                    UNION ALL
                    SELECT "id", "title", NULLIF(TRIM("grandbaby_task_id"), '') FROM {TABLE_FQN}
                    UNION ALL
                    SELECT "id", "title", NULLIF(TRIM("great_grandbaby_task_id"), '') FROM {TABLE_FQN}
                )
                SELECT DISTINCT m.task_id, a.owning_id, a.owning_title
                FROM TEMP_MISSING_HOLIDAY_TASKS m
                LEFT JOIN all_hierarchy a ON a.task_id = m.task_id
                ORDER BY m.task_id
            """
            print("Running the table-wide search (scans the entire Wrike_STG table - may take a bit)...")
            cursor.execute(query)
            rows = cursor.fetchall()
            columns = [desc[0] for desc in cursor.description]

        results = [dict(zip(columns, row)) for row in rows]
        found_elsewhere = [r for r in results if r.get("OWNING_ID") is not None]
        truly_absent = [r for r in results if r.get("OWNING_ID") is None]

        print(f"\nTotal checked: {len(results)}")
        print(f"Found under SOME project in Wrike_STG: {len(found_elsewhere)}")
        print(f"Truly absent anywhere in Wrike_STG:     {len(truly_absent)}")

        if found_elsewhere:
            print("\nSample of tasks found elsewhere, and under which project:")
            for r in found_elsewhere[:15]:
                print(f"  {r['TASK_ID']} -> owning_id={r['OWNING_ID']} ({r['OWNING_TITLE']})")
    finally:
        connection.close()


if __name__ == "__main__":
    main()
