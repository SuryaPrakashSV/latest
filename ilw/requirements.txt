SELECT DISTINCT
    "id" AS entity_id,
    "title" AS entity_name,
    "is_project" AS is_project
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
WHERE "key" = "id"
  AND "title" IN (
      'Intake Form - Resourcing Template',
      'NPI Project Blueprints',
      'Test Import',
      'xxOLD Blueprints'
  )
ORDER BY entity_name, entity_id;
