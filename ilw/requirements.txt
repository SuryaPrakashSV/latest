#!/usr/bin/env python3
"""Confirm the base type of the four named Wrike items using Surya's token.

Save beside check_missing_wrike_ids.py, review_akash_comparison.py and
validate_local_baseline.py in Desktop/wrike-local-baseline, then run:
    python3 confirm_four_wrike_types.py

Only the four exact IDs are checked. Optional filtered lookups are restricted
to the returned permalink. No extraction code is executed and no data is changed
in Wrike or Snowflake. The token is never written to the output files.
"""
import csv
import getpass
import hashlib
import json
import sys
import time
import warnings
from collections import Counter
from datetime import datetime, timezone
from pathlib import Path
from urllib.parse import quote

import requests
import check_missing_wrike_ids as shared
import review_akash_comparison as review
import validate_local_baseline as baseline


BASE_URL = "https://www.wrike.com/api/v4"
LOCAL_EXPORT = "output/local_validation/20260924T213922_287354Z/wrike_local_full.csv"
DOCS = [
    "https://developers.wrike.com/reference/getfoldersmulti",
    "https://developers.wrike.com/reference/getfoldersempty",
]
COLUMNS = [
    "id", "expected_name", "api_title", "type", "reason", "scope",
    "custom_item_type_id", "permalink", "project_field_present", "project_json",
    "direct_http_status", "filter_http_status", "error_code", "checked_at_utc",
    "local_dataset_rows", "local_own_rows", "local_is_project_values",
]
ATTEMPT_COLUMNS = ["time_utc", "id", "step", "attempt", "http_status", "reason", "wait_seconds"]


def now():
    return datetime.now(timezone.utc).isoformat()


def read_local_context(root):
    path = root / LOCAL_EXPORT
    rows = {
        entity_id: {"dataset_rows": 0, "own_rows": 0, "raw_is_project_values": set(), "own_row_names": set()}
        for entity_id in review.EXTRA_CONTAINERS
    }
    if not path.is_file():
        return {}, {"status": "UNAVAILABLE", "path": str(path), "reason": "Saved user export not found; direct check remains available."}
    before, count = baseline.fingerprint(path), 0
    try:
        for row in baseline.read_rows(path, ("id", "key", "title", "is_project")):
            count += 1
            entity_id = row["id"].strip()
            if entity_id not in rows:
                continue
            item = rows[entity_id]
            item["dataset_rows"] += 1
            if row["key"].strip() == entity_id:
                item["own_rows"] += 1
                item["raw_is_project_values"].add(row["is_project"])
                item["own_row_names"].add(row["title"])
        audit = baseline.file_audit(path, before, count)
    except (OSError, ValueError, csv.Error):
        return {}, {"status": "UNAVAILABLE", "path": str(path), "reason": "Saved user export could not be validated; direct check remains available."}
    for item in rows.values():
        item["raw_is_project_values"] = sorted(item["raw_is_project_values"])
        item["own_row_names"] = sorted(item["own_row_names"])
    audit.update({"status": "READ", "interpretation": "Transformed extraction flags are context, not independent proof of API item type."})
    return rows, audit


def parse_response(response, entity_id, allow_empty=False):
    """Accept only a bounded, unpaginated, exact-ID folder response."""
    try:
        body = response.json()
    except ValueError:
        return None, "NON_JSON_RESPONSE", ""
    if not isinstance(body, dict):
        return None, "UNEXPECTED_JSON_SHAPE", ""
    error = body.get("error", "")
    error = error if isinstance(error, str) and len(error) <= 100 and all(c.isalnum() or c == "_" for c in error) else ""
    if response.status_code != 200:
        return None, "HTTP_" + str(response.status_code), error
    if body.get("kind") != "folders" or not isinstance(body.get("data"), list):
        return None, "UNEXPECTED_FOLDER_RESPONSE", error
    if body.get("nextPageToken") or body.get("nextPage"):
        return None, "UNEXPECTED_PAGINATION", error
    data = body["data"]
    if not data and allow_empty:
        return None, "FILTER_RETURNED_NO_MATCH", error
    if len(data) != 1 or not isinstance(data[0], dict) or data[0].get("id") != entity_id:
        return None, "RETURNED_ID_MISMATCH_OR_COUNT", error
    return data[0], "EXACT_ID_RETURNED", error


def valid_project_object(value):
    return (isinstance(value, dict) and isinstance(value.get("ownerIds"), list)
            and all(isinstance(owner, str) for owner in value["ownerIds"])
            and isinstance(value.get("status"), str) and bool(value["status"].strip()))


class Client:
    def __init__(self, session, token, proxies, emit, sleep=time.sleep, monotonic=time.monotonic):
        self.session, self.token, self.proxies, self.emit = session, token, proxies, emit
        self.sleep, self.monotonic, self.next_request = sleep, monotonic, 0.0

    def request(self, entity_id, step, endpoint, params):
        shared.check_id(entity_id)
        for attempt in range(1, shared.MAX_ATTEMPTS + 1):
            wait = max(0.0, self.next_request - self.monotonic())
            if wait:
                self.sleep(wait)
            self.next_request = self.monotonic() + 1.0
            status, error, item, stop = "", "", None, False
            try:
                response = self.session.get(
                    BASE_URL + endpoint, params=params,
                    headers={"Authorization": "Bearer " + self.token},
                    proxies=self.proxies, timeout=(15, 60), allow_redirects=False,
                )
            except KeyboardInterrupt:
                self.emit(entity_id, step, attempt, "", "INTERRUPTED", 0)
                raise
            except requests.RequestException:
                reason, retry, delay = "NETWORK_REQUEST_ERROR", True, float(2 ** (attempt - 1))
            else:
                status = response.status_code
                if status == 401:
                    reason, retry, delay, stop = "AUTHENTICATION_FAILED", False, 0, True
                elif status in shared.RETRY_HTTP:
                    reason, retry = "RATE_LIMITED" if status == 429 else "RETRYABLE_SERVER_ERROR", True
                    delay = shared.retry_wait(response.headers.get("Retry-After"), 5.0 if status == 429 else float(2 ** (attempt - 1)))
                else:
                    item, reason, error = parse_response(response, entity_id, allow_empty=step != "direct")
                    retry, delay = False, 0
            if retry:
                if delay > shared.MAX_WAIT:
                    reason, stop, retry = "RETRY_AFTER_EXCEEDS_WAIT_LIMIT", True, False
                elif attempt == shared.MAX_ATTEMPTS:
                    reason, stop, retry = reason + "_RETRIES_EXHAUSTED", True, False
            self.emit(entity_id, step, attempt, status, reason, delay if retry else 0)
            if retry:
                self.next_request = max(self.next_request, self.monotonic() + delay)
                continue
            return {"http_status": status, "reason": reason, "error_code": error, "item": item}, stop
        raise RuntimeError("Request ended without a recorded result.")


def minimized(item):
    if not item:
        return None
    return {key: item[key] for key in ("id", "title", "scope", "permalink", "customItemTypeId", "project") if key in item}


def classify(client, entity_id, expected_name, local=None):
    local = local or {}
    result = {
        "id": entity_id, "expected_name": expected_name, "api_title": "", "type": "UNRESOLVED",
        "reason": "", "scope": "", "custom_item_type_id": "", "permalink": "",
        "project_field_present": "", "project_json": "", "direct_http_status": "",
        "filter_http_status": "", "error_code": "", "checked_at_utc": now(),
        "local_dataset_rows": local.get("dataset_rows", ""), "local_own_rows": local.get("own_rows", ""),
        "local_is_project_values": json.dumps(local.get("raw_is_project_values", []), ensure_ascii=False),
    }
    evidence = {"id": entity_id, "direct": None, "filtered": None, "filter_project": None}
    direct, stop = client.request(entity_id, "direct", "/folders/" + quote(entity_id, safe=""), {"fields": '["customItemTypeId"]'})
    item = direct["item"]
    result.update({"direct_http_status": direct["http_status"], "reason": direct["reason"], "error_code": direct["error_code"]})
    evidence["direct"] = {**direct, "item": minimized(item)}
    if item is None:
        return result, evidence, stop
    project, scope = item.get("project"), item.get("scope", "")
    result.update({
        "api_title": item.get("title", ""), "scope": scope,
        "custom_item_type_id": item.get("customItemTypeId", ""), "permalink": item.get("permalink", ""),
        "project_field_present": int("project" in item),
        "project_json": json.dumps(project, ensure_ascii=False) if "project" in item else "",
    })
    if not isinstance(scope, str):
        result["reason"] = "MALFORMED_SCOPE"
        return result, evidence, False
    if project is not None and not isinstance(project, dict):
        result["reason"] = "MALFORMED_PROJECT_FIELD"
        return result, evidence, False
    if valid_project_object(project):
        result.update({"type": "PROJECT", "reason": "PROJECT_OBJECT_RETURNED"})
        return result, evidence, False
    permalink = item.get("permalink")
    if not isinstance(permalink, str) or not permalink.strip():
        result["reason"] = "PERMALINK_REQUIRED_TO_CONFIRM_BASE_TYPE"
        return result, evidence, False
    expected_project = isinstance(project, dict)
    params = {"permalink": permalink, "project": "true" if expected_project else "false", "pageSize": 2,
              "fields": '["customItemTypeId"]'}
    if scope == "RbFolder":
        params["deleted"] = "true"
    evidence["filter_project"] = expected_project
    filtered, stop = client.request(entity_id, "project_filter" if expected_project else "folder_filter", "/folders", params)
    match = filtered["item"]
    evidence["filtered"] = {**filtered, "item": minimized(match)}
    result.update({"filter_http_status": filtered["http_status"], "reason": filtered["reason"], "error_code": filtered["error_code"]})
    if match is None:
        return result, evidence, stop
    if match.get("permalink") != permalink:
        result["reason"] = "FILTER_PERMALINK_MISMATCH"
        return result, evidence, False
    if match.get("scope", scope) != scope:
        result["reason"] = "SCOPE_CHANGED_DURING_CHECK"
        return result, evidence, False
    match_project = match.get("project")
    if expected_project:
        if "project" in match and not isinstance(match_project, dict):
            result["reason"] = "PROJECT_TRUE_FILTER_CONFLICT"
        else:
            result.update({"type": "PROJECT", "reason": "MATCHED_EXACT_PERMALINK_PROJECT_TRUE_FILTER"})
    elif match_project is not None:
        result["reason"] = "PROJECT_FALSE_FILTER_CONFLICT"
    else:
        result.update({"type": "FOLDER" if scope == "WsFolder" else "NON_PROJECT_CONTAINER",
                       "reason": "MATCHED_EXACT_PERMALINK_PROJECT_FALSE_FILTER"})
    return result, evidence, stop


def write_outputs(destination, report):
    shared.write_summary(destination / "type_evidence.json", report)
    temporary = destination / "four_item_types.csv.tmp"
    with temporary.open("w", encoding="utf-8-sig", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=COLUMNS)
        writer.writeheader()
        writer.writerows(report["results"])
    temporary.replace(destination / "four_item_types.csv")


def run(root, token, proxies, source_audit, local, local_audit, session=None):
    destination = root / "output" / "four_item_types" / datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%S_%fZ")
    destination.mkdir(parents=True, exist_ok=False)
    report = {
        "started_at_utc": now(), "status": "RUNNING", "token_label": "Surya/production token supplied by operator; identity not independently verified",
        "proxy_source": source_audit, "local_export": local_audit, "local_context": local,
        "helper_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "documentation": DOCS, "results": [], "evidence": [],
        "interpretation": "Base type as returned at check time. Custom item type IDs and scopes are retained. A failed lookup leaves type unresolved. This does not verify permissions, historical type or task completeness.",
    }
    write_outputs(destination, report)
    own_session = session is None
    session = requests.Session() if session is None else session
    completed, stopped, active_id = set(), False, None
    try:
        with (destination / "attempts.csv").open("x", encoding="utf-8-sig", newline="") as handle:
            writer = csv.DictWriter(handle, fieldnames=ATTEMPT_COLUMNS)
            writer.writeheader()
            handle.flush()
            def emit(entity_id, step, attempt, status, reason, delay):
                writer.writerow(dict(zip(ATTEMPT_COLUMNS, (now(), entity_id, step, attempt, status, reason, delay))))
                handle.flush()
            client = Client(session, token, proxies, emit)
            for entity_id, expected_name in review.EXTRA_CONTAINERS.items():
                active_id = entity_id
                result, evidence, stop = classify(client, entity_id, expected_name, local.get(entity_id))
                report["results"].append(result)
                report["evidence"].append(evidence)
                completed.add(entity_id)
                write_outputs(destination, report)
                print("{}: {} ({})".format(expected_name, result["type"], result["reason"]), flush=True)
                if stop:
                    stopped = True
                    report["stop_reason"] = result["reason"]
                    break
    except KeyboardInterrupt:
        stopped = True
        report["stop_reason"] = "INTERRUPTED"
    except Exception as exc:
        stopped = True
        report["stop_reason"] = "LOCAL_ERROR_" + type(exc).__name__
    finally:
        if own_session:
            session.close()
        if active_id is not None and active_id not in completed:
            report["results"].append({
                "id": active_id, "expected_name": review.EXTRA_CONTAINERS[active_id],
                "type": "UNRESOLVED", "reason": report.get("stop_reason", "INTERRUPTED_BEFORE_RESULT"),
                "checked_at_utc": now(),
            })
            completed.add(active_id)
        for entity_id, expected_name in review.EXTRA_CONTAINERS.items():
            if entity_id not in completed:
                report["results"].append({"id": entity_id, "expected_name": expected_name, "type": "NOT_CHECKED", "reason": "CHECK_STOPPED_BEFORE_RESULT"})
        report.update({"status": "STOPPED" if stopped else "FINISHED", "finished_at_utc": now(),
                       "counts": dict(Counter(row["type"] for row in report["results"]))})
        write_outputs(destination, report)
    print("Type counts: " + json.dumps(report["counts"], sort_keys=True), flush=True)
    print("Saved: " + str(destination), flush=True)
    return 0 if all(row["type"] in {"PROJECT", "FOLDER", "NON_PROJECT_CONTAINER"} for row in report["results"]) else 2


def main():
    root = Path(__file__).resolve().parent
    for entity_id in review.EXTRA_CONTAINERS:
        shared.check_id(entity_id)
    proxies, source_audit = shared.read_proxies(root / "Wrike_Data_local_validation.py")
    local, audit = read_local_context(root)
    if local:
        for entity_id, item in local.items():
            print("Saved export - {}: own rows {}; is_project values {}".format(review.EXTRA_CONTAINERS[entity_id], item["own_rows"], repr(item["raw_is_project_values"])))
    else:
        print(audit["reason"])
    print("Checking only four IDs. Enter YOUR production Wrike token, not Akash's.", flush=True)
    with warnings.catch_warnings():
        warnings.simplefilter("error", getpass.GetPassWarning)
        token = getpass.getpass("Your Wrike token (hidden): ").strip()
    if not token or any(char.isspace() for char in token):
        raise ValueError("Token is empty or contains whitespace; no calls made.")
    return run(root, token, proxies, source_audit, local, audit)


if __name__ == "__main__":
    try:
        sys.exit(main())
    except (OSError, ValueError, csv.Error, getpass.GetPassWarning) as exc:
        print("Could not complete type check ({}). Check that this file and its helper files are saved beside Wrike_Data_local_validation.py and use an interactive terminal.".format(type(exc).__name__), file=sys.stderr)
        sys.exit(1)
