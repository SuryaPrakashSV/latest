SELECT
    "is_project"::VARCHAR AS is_project_value,
    COUNT(*) AS total_rows,
    COUNT_IF("key" = "id") AS key_equals_id_rows,
    COUNT_IF(
        NULLIF(TRIM("key"::VARCHAR), '') IS NULL
    ) AS missing_key_rows,
    MAX(
        TRY_TO_TIMESTAMP_TZ("updatedDate"::VARCHAR)
    ) AS most_recent_update
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
GROUP BY "is_project"::VARCHAR
ORDER BY total_rows DESC;
