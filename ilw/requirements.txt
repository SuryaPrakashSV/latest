import csv

with open("akash_reference.csv", encoding="utf-8-sig", newline="") as f:
    reader = csv.DictReader(f)
    reader.fieldnames = [
        name.strip().lower() for name in (reader.fieldnames or [])
    ]
    if "id" not in reader.fieldnames:
        raise SystemExit("Missing id column. Found: " + str(reader.fieldnames))
    rows = list(reader)

ids = [(row.get("id") or "").strip() for row in rows]
valid_ids = [value for value in ids if value]

print("Data rows:", len(rows))
print("Unique IDs:", len(set(valid_ids)))
print("Blank ID rows:", len(rows) - len(valid_ids))
print("Repeated ID rows:", len(valid_ids) - len(set(valid_ids)))
PY
