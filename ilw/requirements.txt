from pathlib import Path
import pandas as pd


# =========================================================
# FILES
# =========================================================

BASE_DIR = Path(__file__).resolve().parent

MY_FILE = BASE_DIR / "Single_execut_my_token.xlsx"
AKASH_FILE = BASE_DIR / "Single_execut_akash_token.xlsx"

OUTPUT_FILE = BASE_DIR / "SDP.xlsx"


# =========================================================
# COLUMNS NEEDED FROM SINGLE_EXECUT
# =========================================================

REQUIRED_COLUMNS = [
    "id",
    "key",
    "title",
    "is_project",
    "parent_folder_id",
    "parent_folder_title",
    "child_folder_id",
    "child_folder_title",
    "grandchild_folder_id",
    "grandchild_folder_title",
    "baby_folder_id",
    "baby_folder_title",
    "grandbaby_folder_id",
    "grandbaby_folder_title",
]


# =========================================================
# HELPER
# =========================================================

def clean(value):

    if pd.isna(value):
        return ""

    value = str(value).strip()

    if value.lower() in {
        "",
        "nan",
        "none",
        "null",
        "<na>",
    }:
        return ""

    return value


# =========================================================
# LOAD PROJECT IDS + PROJECT NAMES
# =========================================================

def load_projects(path, token_label):

    print("\n" + "=" * 70)
    print(f"Loading {path.name} ...")
    print("=" * 70)

    if not path.exists():
        raise FileNotFoundError(
            f"File not found: {path}"
        )

    # Only load the columns needed for this comparison.
    df = pd.read_excel(
        path,
        dtype=str,
        engine="openpyxl",
        usecols=lambda c: str(c).strip() in REQUIRED_COLUMNS
    ).fillna("")

    df.columns = [
        str(c).strip()
        for c in df.columns
    ]

    # Required columns must exist.
    for col in ["id", "key", "title", "is_project"]:
        if col not in df.columns:
            raise ValueError(
                f"Required column '{col}' "
                f"not found in {path.name}"
            )

    # Add hierarchy columns if any are absent.
    for col in REQUIRED_COLUMNS:
        if col not in df.columns:
            df[col] = ""

    # Clean values.
    for col in REQUIRED_COLUMNS:
        df[col] = df[col].map(clean)

    df["is_project"] = (
        df["is_project"]
        .str.upper()
    )

    # =====================================================
    # STEP 1
    # IDENTIFY EVERY PROJECT ID VISIBLE TO THIS TOKEN
    # =====================================================

    project_rows = df[
        (df["is_project"] == "Y")
        & (df["id"] != "")
        & (df["id"].str.upper() != "PLACEHOLDER")
    ].copy()

    project_ids = sorted(
        project_rows["id"].unique()
    )

    print(
        f"{token_label} unique project IDs: "
        f"{len(project_ids):,}"
    )

    # =====================================================
    # STEP 2
    # GET PROJECT NAMES FROM ACTUAL PROJECT ROWS
    #
    # Real project/folder row:
    #     id  = project id
    #     key = project id
    #     title = project name
    #
    # Task rows:
    #     id  = folder/project id
    #     key = task id
    # =====================================================

    actual_project_rows = df[
        (df["is_project"] == "Y")
        & (df["id"] != "")
        & (df["id"].str.upper() != "PLACEHOLDER")
        & (df["id"] == df["key"])
        & (df["title"] != "")
    ].copy()

    name_lookup = {}

    # If a project appears multiple times, keep the first valid
    # project name for that ID.
    for _, row in actual_project_rows.iterrows():

        project_id = clean(row["id"])
        project_name = clean(row["title"])

        if (
            project_id
            and project_name
            and project_id not in name_lookup
        ):
            name_lookup[project_id] = project_name

    # =====================================================
    # STEP 3
    # FALLBACK TO FOLDER HIERARCHY FOR ANY PROJECT
    # WHOSE NAME WAS NOT FOUND ABOVE
    # =====================================================

    hierarchy_levels = [
        (
            "parent_folder_id",
            "parent_folder_title"
        ),
        (
            "child_folder_id",
            "child_folder_title"
        ),
        (
            "grandchild_folder_id",
            "grandchild_folder_title"
        ),
        (
            "baby_folder_id",
            "baby_folder_title"
        ),
        (
            "grandbaby_folder_id",
            "grandbaby_folder_title"
        ),
    ]

    missing_ids = set(
        project_id
        for project_id in project_ids
        if project_id not in name_lookup
    )

    for id_col, title_col in hierarchy_levels:

        if not missing_ids:
            break

        matching_rows = df[
            df[id_col].isin(missing_ids)
            & (df[title_col] != "")
        ][
            [
                id_col,
                title_col
            ]
        ].drop_duplicates()

        for _, row in matching_rows.iterrows():

            project_id = clean(
                row[id_col]
            )

            project_name = clean(
                row[title_col]
            )

            if (
                project_id
                and project_name
                and project_id not in name_lookup
            ):
                name_lookup[
                    project_id
                ] = project_name

        missing_ids = set(
            project_id
            for project_id in project_ids
            if project_id not in name_lookup
        )

    # =====================================================
    # STEP 4
    # BUILD FINAL PROJECT ID / NAME TABLE
    # =====================================================

    records = []

    for project_id in project_ids:

        records.append(
            {
                "project_id": project_id,
                "project_name": name_lookup.get(
                    project_id,
                    "[NAME NOT FOUND]"
                )
            }
        )

    projects = pd.DataFrame(
        records
    )

    projects = (
        projects
        .sort_values(
            [
                "project_name",
                "project_id"
            ]
        )
        .reset_index(drop=True)
    )

    missing_name_count = int(
        (
            projects["project_name"]
            == "[NAME NOT FOUND]"
        ).sum()
    )

    print(
        f"{token_label} projects with names: "
        f"{len(projects) - missing_name_count:,}"
    )

    print(
        f"{token_label} names not found: "
        f"{missing_name_count:,}"
    )

    return projects


# =========================================================
# LOAD BOTH TOKEN OUTPUTS
# =========================================================

my_projects = load_projects(
    MY_FILE,
    "MY TOKEN"
)

akash_projects = load_projects(
    AKASH_FILE,
    "AKASH TOKEN"
)


# =========================================================
# CREATE PROJECT ID SETS
# =========================================================

my_ids = set(
    my_projects["project_id"]
)

akash_ids = set(
    akash_projects["project_id"]
)

common_ids = (
    my_ids
    & akash_ids
)

only_my_ids = (
    my_ids
    - akash_ids
)

only_akash_ids = (
    akash_ids
    - my_ids
)


# =========================================================
# FILTER UNIQUE PROJECTS
# =========================================================

only_my_projects = (
    my_projects[
        my_projects[
            "project_id"
        ].isin(
            only_my_ids
        )
    ]
    .copy()
    .sort_values(
        [
            "project_name",
            "project_id"
        ]
    )
    .reset_index(drop=True)
)


only_akash_projects = (
    akash_projects[
        akash_projects[
            "project_id"
        ].isin(
            only_akash_ids
        )
    ]
    .copy()
    .sort_values(
        [
            "project_name",
            "project_id"
        ]
    )
    .reset_index(drop=True)
)


# =========================================================
# CREATE EXACT 8-COLUMN OUTPUT
# =========================================================

output = pd.DataFrame({

    # 1
    "my_token_project_ids":
        pd.Series(
            my_projects[
                "project_id"
            ].tolist()
        ),

    # 2
    "my_token_project_names":
        pd.Series(
            my_projects[
                "project_name"
            ].tolist()
        ),

    # 3
    "akash_token_project_ids":
        pd.Series(
            akash_projects[
                "project_id"
            ].tolist()
        ),

    # 4
    "akash_token_project_names":
        pd.Series(
            akash_projects[
                "project_name"
            ].tolist()
        ),

    # 5
    "project_ids_only_in_my_token":
        pd.Series(
            only_my_projects[
                "project_id"
            ].tolist()
        ),

    # 6
    "project_names_only_in_my_token":
        pd.Series(
            only_my_projects[
                "project_name"
            ].tolist()
        ),

    # 7
    "project_ids_only_in_akash_token":
        pd.Series(
            only_akash_projects[
                "project_id"
            ].tolist()
        ),

    # 8
    "project_names_only_in_akash_token":
        pd.Series(
            only_akash_projects[
                "project_name"
            ].tolist()
        ),
})


# =========================================================
# SAVE SDP.XLSX
# =========================================================

output.to_excel(
    OUTPUT_FILE,
    index=False,
    engine="openpyxl"
)
