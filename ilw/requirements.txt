SELECT
    "id" AS project_id,
    MAX(TRIM("title")) AS project_name,
    CASE
        WHEN MAX(
            CASE
                WHEN UPPER(TRIM("is_project"::VARCHAR)) = 'Y'
                THEN 1 ELSE 0
            END
        ) = 1 THEN 'PROJECT'
        ELSE 'FOLDER_OR_OTHER'
    END AS entity_type
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
WHERE "key" = "id"
  AND NULLIF(TRIM("id"), '') IS NOT NULL
  AND UPPER(TRIM("id")) NOT IN ('PLACEHOLDER', 'NULL')
  AND NULLIF(TRIM("title"), '') IS NOT NULL
  AND UPPER(TRIM("title")) <> 'NULL'
GROUP BY "id"
ORDER BY project_name, project_id;
