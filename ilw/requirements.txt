WITH top_task AS (
    SELECT
        COALESCE(
            NULLIF("great_grandbaby_task_id", ''),
            NULLIF("grandbaby_task_id", ''),
            NULLIF("baby_task_id", ''),
            NULLIF("grandchild_task_id", ''),
            NULLIF("child_task_id", ''),
            NULLIF("parent_task_id", '')
        ) AS task_id,
        COUNT(*) AS row_count
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
    WHERE "id" = 'MQAAAABnIKFK'
      AND COALESCE(
            NULLIF("great_grandbaby_task_id", ''),
            NULLIF("grandbaby_task_id", ''),
            NULLIF("baby_task_id", ''),
            NULLIF("grandchild_task_id", ''),
            NULLIF("child_task_id", ''),
            NULLIF("parent_task_id", '')
          ) IS NOT NULL
    GROUP BY 1
    ORDER BY row_count DESC
    LIMIT 1
)

SELECT
    COALESCE(
        NULLIF("great_grandbaby_task_id", ''),
        NULLIF("grandbaby_task_id", ''),
        NULLIF("baby_task_id", ''),
        NULLIF("grandchild_task_id", ''),
        NULLIF("child_task_id", ''),
        NULLIF("parent_task_id", '')
    ) AS task_id,
    "status",
    "customStatusId",
    "createdDate",
    "updatedDate",
    "completedDate"
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
WHERE "id" = 'MQAAAABnIKFK'
  AND COALESCE(
        NULLIF("great_grandbaby_task_id", ''),
        NULLIF("grandbaby_task_id", ''),
        NULLIF("baby_task_id", ''),
        NULLIF("grandchild_task_id", ''),
        NULLIF("child_task_id", ''),
        NULLIF("parent_task_id", '')
    ) = (SELECT task_id FROM top_task)
ORDER BY "updatedDate";
