from pathlib import Path
import pandas as pd


# =========================================================
# FILES
# =========================================================

BASE_DIR = Path(__file__).resolve().parent

MY_FILE = BASE_DIR / "Single_execut_my_token.xlsx"
AKASH_FILE = BASE_DIR / "Single_execut_akash_token.xlsx"

PROJECT_NAMES_FILE = BASE_DIR / "project_names.csv"

OUTPUT_FILE = BASE_DIR / "token_project_comparison.xlsx"


# =========================================================
# CLEANING
# =========================================================

def clean(value):
    if pd.isna(value):
        return ""

    value = str(value).strip()

    if value.lower() in {
        "",
        "nan",
        "none",
        "null",
        "<na>",
    }:
        return ""

    return value


# =========================================================
# LOAD PROJECT IDs FROM EACH TOKEN OUTPUT
# =========================================================

def load_project_ids(path):

    print(f"\nLoading {path.name} ...")

    df = pd.read_excel(
        path,
        dtype=str,
        engine="openpyxl",
        usecols=["id", "is_project"]
    ).fillna("")

    df["id"] = df["id"].map(clean)

    df["is_project"] = (
        df["is_project"]
        .map(clean)
        .str.upper()
    )

    projects = df[
        (df["is_project"] == "Y")
        & (df["id"] != "")
        & (df["id"].str.upper() != "PLACEHOLDER")
    ]

    project_ids = sorted(
        set(projects["id"])
    )

    print(
        f"Unique project IDs: "
        f"{len(project_ids):,}"
    )

    return project_ids


# =========================================================
# LOAD PROJECT ID -> NAME LOOKUP
# =========================================================

def load_project_names():

    print(
        f"\nLoading project names from "
        f"{PROJECT_NAMES_FILE.name} ..."
    )

    names = pd.read_csv(
        PROJECT_NAMES_FILE,
        dtype=str
    ).fillna("")

    names.columns = [
        str(c).strip()
        for c in names.columns
    ]

    required = {
        "project_api_id",
        "project_name"
    }

    missing = required - set(names.columns)

    if missing:
        raise ValueError(
            "project_names.csv is missing columns: "
            f"{sorted(missing)}"
        )

    names["project_api_id"] = (
        names["project_api_id"]
        .map(clean)
    )

    names["project_name"] = (
        names["project_name"]
        .map(clean)
    )

    names = names[
        names["project_api_id"] != ""
    ]

    names = names.drop_duplicates(
        subset=["project_api_id"]
    )

    lookup = dict(
        zip(
            names["project_api_id"],
            names["project_name"]
        )
    )

    print(
        f"Project names loaded: "
        f"{len(lookup):,}"
    )

    return lookup


# =========================================================
# LOAD DATA
# =========================================================

my_ids = load_project_ids(MY_FILE)

akash_ids = load_project_ids(AKASH_FILE)

name_lookup = load_project_names()


# =========================================================
# COMPARE PROJECT IDs
# =========================================================

my_set = set(my_ids)

akash_set = set(akash_ids)

common_ids = my_set & akash_set

only_my_ids = sorted(
    my_set - akash_set
)

only_akash_ids = sorted(
    akash_set - my_set
)


# =========================================================
# CREATE ID + NAME PAIRS
# =========================================================

def make_project_list(project_ids):

    rows = []

    for project_id in project_ids:

        project_name = name_lookup.get(
            project_id,
            "[NAME NOT FOUND]"
        )

        rows.append(
            {
                "project_id": project_id,
                "project_name": project_name
            }
        )

    # Sort by project name
    rows = sorted(
        rows,
        key=lambda x: (
            x["project_name"].lower(),
            x["project_id"]
        )
    )

    return rows


my_projects = make_project_list(
    my_ids
)

akash_projects = make_project_list(
    akash_ids
)

only_my_projects = make_project_list(
    only_my_ids
)

only_akash_projects = make_project_list(
    only_akash_ids
)


# =========================================================
# BUILD SINGLE EXCEL TABLE
# =========================================================

output = pd.DataFrame({

    "my_token_project_id":
        pd.Series([
            x["project_id"]
            for x in my_projects
        ]),

    "my_token_project_name":
        pd.Series([
            x["project_name"]
            for x in my_projects
        ]),

    "akash_token_project_id":
        pd.Series([
            x["project_id"]
            for x in akash_projects
        ]),

    "akash_token_project_name":
        pd.Series([
            x["project_name"]
            for x in akash_projects
        ]),

    "only_my_token_project_id":
        pd.Series([
            x["project_id"]
            for x in only_my_projects
        ]),

    "only_my_token_project_name":
        pd.Series([
            x["project_name"]
            for x in only_my_projects
        ]),

    "only_akash_token_project_id":
        pd.Series([
            x["project_id"]
            for x in only_akash_projects
        ]),

    "only_akash_token_project_name":
        pd.Series([
            x["project_name"]
            for x in only_akash_projects
        ]),
})


# =========================================================
# SAVE EXCEL
# =========================================================

output.to_excel(
    OUTPUT_FILE,
    index=False,
    engine="openpyxl"
)


# =========================================================
# TERMINAL SUMMARY
# =========================================================

print("\n" + "=" * 70)
print("PROJECT TOKEN COMPARISON")
print("=" * 70)

print(
    f"My token projects       : "
    f"{len(my_ids):,}"
)

print(
    f"Akash token projects    : "
    f"{len(akash_ids):,}"
)

print(
    f"Common projects         : "
    f"{len(common_ids):,}"
)

print(
    f"Only in my token        : "
    f"{len(only_my_ids):,}"
)

print(
    f"Only in Akash token     : "
    f"{len(only_akash_ids):,}"
)

my_missing_names = sum(
    1
    for x in my_projects
    if x["project_name"] == "[NAME NOT FOUND]"
)

akash_missing_names = sum(
    1
    for x in akash_projects
    if x["project_name"] == "[NAME NOT FOUND]"
)

print(
    f"My IDs without names    : "
    f"{my_missing_names:,}"
)

print(
    f"Akash IDs without names : "
    f"{akash_missing_names:,}"
)

print("\nComparison complete.")
print(f"Excel created: {OUTPUT_FILE}")
