SELECT COLUMN_NAME
FROM GBI_RETAIL_BAP_DB.INFORMATION_SCHEMA.COLUMNS
WHERE UPPER(TABLE_SCHEMA) = 'ASO_OPS_WSA_SNDBX_BIZ_APP'
  AND UPPER(TABLE_NAME)   = 'WRIKE_STG'
  AND (LOWER(COLUMN_NAME) LIKE '%title%' OR LOWER(COLUMN_NAME) LIKE '%name%')
ORDER BY ORDINAL_POSITION;



SELECT project_api_id, project_name
FROM (
    SELECT
        "id"                                          AS project_api_id,
        MAX(CASE WHEN "key" = "id" THEN "title" END)  AS project_name
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
    WHERE NULLIF(TRIM("id"), '') IS NOT NULL
      AND UPPER(TRIM("id")) <> 'PLACEHOLDER'
    GROUP BY "id"
)
WHERE project_name IS NOT NULL
ORDER BY project_name;
