def status_workflow_map(client) -> dict[str, dict]:
    """customStatusId -> {"workflow_id", "workflow_name"}, built from GET /workflows.

    Reuses flatten_workflows() (already used by map-statuses) instead of a
    second parallel implementation - one live call, one source of truth for
    which workflow a status belongs to.
    """
    from wrike_status_api import flatten_workflows

    catalog = flatten_workflows(client.workflows_payload())
    return {
        row["custom_status_id"]: {"workflow_id": row["workflow_id"], "workflow_name": row["workflow_name"]}
        for row in catalog
    }


def parse_expected(pairs: list[str] | None) -> list[dict[str, str]]:
    """Parse repeated --expected "Status Name:YYYY-MM-DD" values.

    Kept separate from argparse wiring so it's testable on its own - a typo
    here (missing colon, bad date) should fail loudly, not silently produce
    an expectation nobody meant to check.
    """
    parsed: list[dict[str, str]] = []
    for pair in pairs or []:
        if ":" not in pair:
            raise SystemExit(
                f"--expected value {pair!r} must be 'Status Name:YYYY-MM-DD' (missing ':')"
            )
        status_name, _, date_part = pair.partition(":")
        status_name = status_name.strip()
        date_part = date_part.strip()
        if not status_name or not date_part:
            raise SystemExit(f"--expected value {pair!r} must be 'Status Name:YYYY-MM-DD'")
        try:
            pd.Timestamp(date_part)
        except (ValueError, TypeError) as exc:
            raise SystemExit(f"--expected value {pair!r} has an invalid date: {exc}")
        parsed.append({"status_name": status_name, "status_changed_date": date_part})
    return parsed


def token() -> str:
    value = os.getenv("WRIKE_TOKEN", "").strip()
    return value or getpass.getpass("Wrike token (hidden; not saved): ").strip()


def callback_url() -> str:
    value = os.getenv("WRIKE_WEBHOOK_URL", "").strip()
    return value or getpass.getpass("Approved callback URL (hidden; not saved): ").strip()


def webhook_secret() -> str:
    value = os.getenv("WRIKE_WEBHOOK_SECRET", "").strip()
    return value or getpass.getpass("Webhook signing secret (hidden; not saved): ").strip()


def redact_webhook_response(value):
    """Remove callback URLs and secrets before writing registration evidence."""
    if isinstance(value, dict):
        return {
            key: "[REDACTED]" if key.lower() in {"hookurl", "secret"}
            else redact_webhook_response(item)
            for key, item in value.items()
        }
    if isinstance(value, list):
        return [redact_webhook_response(item) for item in value]
    return value


def dated_output_dir(base: str) -> Path:
    """Namespace a diagnostic output directory by UTC date (base/YYYYMMDD/).

    A rerun on a later day writes to a new folder instead of silently
    overwriting the previous day's evidence. These are point-in-time
    snapshots and cannot be recreated once lost — see the deleted UAT
    tracker copies incident (2026-08-27), which is exactly this failure
    mode and is why this exists.
    """
    stamp = pd.Timestamp.now(tz="UTC").strftime("%Y%m%d")
    path = Path(base) / stamp
    path.mkdir(parents=True, exist_ok=True)
    return path


def write_outputs(history: pd.DataFrame, output: Path, diagnostics: dict | None = None) -> None:
    output.mkdir(parents=True, exist_ok=True)
    history.to_csv(output / "project_status_history.csv", index=False)
    lifecycle_metrics(history).to_csv(output / "project_status_lifecycle.csv", index=False)
    if diagnostics is not None:
        (output / "diagnostics.json").write_text(json.dumps(diagnostics, indent=2, default=str))
    print(f"Wrote {len(history)} history rows to {output}")

        separators=(",", ":"),

        (
            "audit_log",

                "operations": json.dumps(["TaskStatusChanged"], separators=(",", ":")),

                    for sub_id in (task.get("subTaskIds") or [])
                    if str(sub_id) not in tasks_by_id
                }
            )
            if not missing:
                break
            for fetched in client.tasks_batch(missing):
                if fetched.get("id"):
                    tasks_by_id[str(fetched["id"])] = fetched

        # the project-level 403 automatically applies to tasks too.
        try:
            client.audit_log(object_ids=[args.project_id], start="2000-01-01T00:00:00Z")
            audit_log_blocked_reason = None
        except WrikePermissionError as exc:
            audit_log_blocked_reason = str(exc)

            expected = pd.read_csv(args.expected_csv, low_memory=False)
            if "task_id" not in expected.columns:
                raise SystemExit(
                    f"--expected-csv {args.expected_csv} has no 'task_id' column; "
                    f"available columns: {list(expected.columns)}"
                )
            expected_ids = expected["task_id"].astype(str).tolist()

        classification = classify_task_history_availability(
            tasks_by_id=tasks_by_id,
            status_names=status_names,
            expected_task_ids=expected_ids,
            audit_log_blocked_reason=audit_log_blocked_reason,
        )

        output = dated_output_dir(args.output)
        classification.to_csv(output / "task_history_classification.csv", index=False)

        counts = classification["classification"].value_counts().to_dict()
        print(f"Classified {len(classification)} task(s)/subtask(s) under {args.project_id}:")
        for label in sorted(counts):
            print(f"  {label}: {counts[label]}")
        if audit_log_blocked_reason:
            print(f"\nAudit Log attempt: BLOCKED - {audit_log_blocked_reason}")
        else:
            print("\nAudit Log attempt: reachable - re-examine any non-CURRENT_STATUS_ONLY rows above.")
        print(f"\nWrote {output}/task_history_classification.csv")
