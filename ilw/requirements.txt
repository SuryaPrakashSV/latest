"""
Quick one-task probe.

Purpose:
Make one plain GET /tasks/{id} request with NO explicit `fields` parameter
and inspect which hierarchy-related fields Wrike returns by default.

Usage:
    python3 probe_fields.py

The script will:
1. Ask for the Wrike token using a hidden prompt if WRIKE_TOKEN is not set.
2. Ask for one real task ID.
3. Make one GET /tasks/{task_id} request.
4. Print every key returned by Wrike.
5. Specifically show whether the hierarchy fields we care about are present.
"""

import getpass
import os

import requests

API_BASE = "https://www.wrike.com/api/v4"


def token() -> str:
    value = os.getenv("WRIKE_TOKEN", "").strip()

    if value:
        return value

    return getpass.getpass(
        "Wrike token (hidden; not saved): "
    ).strip()


def main() -> None:
    tok = token()

    headers = {
        "Authorization": f"Bearer {tok}"
    }

    task_id = input(
        "Paste one real task ID to probe against: "
    ).strip()

    if not task_id:
        raise SystemExit("Task ID cannot be blank.")

    print(f"\nFetching default response for task: {task_id}\n")

    resp = requests.get(
        f"{API_BASE}/tasks/{task_id}",
        headers=headers,
        timeout=30,
    )

    if not resp.ok:
        print(f"Request failed: HTTP {resp.status_code}")
        print(resp.text)
        return

    payload = resp.json()
    data = payload.get("data", [])

    if not data:
        print("Request succeeded, but Wrike returned no task data.")
        return

    task = data[0]

    print("All keys in the default response:")
    print(sorted(task.keys()))

    print("\nHierarchy / task fields:\n")

    fields_to_check = (
        "id",
        "title",
        "parentIds",
        "superTaskIds",
        "superParentIds",
        "subTaskIds",
    )

    for field in fields_to_check:
        if field in task:
            print(f"{field}: PRESENT -> {task.get(field)}")
        else:
            print(f"{field}: ABSENT")


if __name__ == "__main__":
    main()
