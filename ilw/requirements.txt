WITH top_task AS (
    SELECT
        COALESCE(
            "great_grandbaby_task_id",
            "grandbaby_task_id",
            "baby_task_id",
            "grandchild_task_id",
            "child_task_id",
            "parent_task_id"
        ) AS task_id,
        COUNT(*) AS row_count
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
    WHERE "id" = 'MQAAAABnIKFK'
      AND COALESCE(
            "great_grandbaby_task_id",
            "grandbaby_task_id",
            "baby_task_id",
            "grandchild_task_id",
            "child_task_id",
            "parent_task_id"
          ) IS NOT NULL
    GROUP BY 1
    ORDER BY row_count DESC
    LIMIT 1
)

SELECT
    COALESCE(
        "great_grandbaby_task_id",
        "grandbaby_task_id",
        "baby_task_id",
        "grandchild_task_id",
        "child_task_id",
        "parent_task_id"
    ) AS task_id,
    "status",
    "customStatusId",
    "createdDate",
    "updatedDate",
    "completedDate"
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
WHERE COALESCE(
        "great_grandbaby_task_id",
        "grandbaby_task_id",
        "baby_task_id",
        "grandchild_task_id",
        "child_task_id",
        "parent_task_id"
    ) = (SELECT task_id FROM top_task)
ORDER BY "updatedDate";
