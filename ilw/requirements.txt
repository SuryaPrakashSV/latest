SELECT
    TABLE_NAME,
    COLUMN_NAME,
    DATA_TYPE
FROM GBI_RETAIL_BAP_DB.INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'ASO_OPS_WSA_SNDBX_BIZ_APP'
  AND TABLE_NAME IN (
      'WRIKE_API_BAU_PROJECT_LEVEL',
      'WRIKE_TEST',
      'WRIKE_PROJECT_STATUS_HISTORY_SCD'
  )
ORDER BY TABLE_NAME, ORDINAL_POSITION;
