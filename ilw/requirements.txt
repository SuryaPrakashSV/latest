SELECT "is_project", COUNT(*) AS rows, COUNT(DISTINCT "id") AS distinct_ids
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
GROUP BY "is_project";



SELECT DISTINCT "id" AS project_id, "title" AS project_name
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
WHERE "is_project" = TRUE
  AND NULLIF(TRIM("id"), '') IS NOT NULL
  AND UPPER(TRIM("id")) <> 'PLACEHOLDER'
ORDER BY project_name;
