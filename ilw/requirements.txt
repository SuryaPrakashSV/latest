#!/usr/bin/env python3
"""Compare saved Wrike scripts without running them or printing source values."""
import argparse
import ast
import difflib
import hashlib
import io
from pathlib import Path
import re
import tokenize


def read_source(path):
    before = path.stat()
    raw = path.read_bytes()
    after = path.stat()
    if (before.st_size, before.st_mtime_ns) != (after.st_size, after.st_mtime_ns):
        raise ValueError("A source file changed during the check; rerun after saving.")
    encoding, _ = tokenize.detect_encoding(io.BytesIO(raw).readline)
    source = raw.decode(encoding)
    try:
        tree = ast.parse(source)
    except SyntaxError as error:
        raise ValueError(f"Syntax error at line {error.lineno}; source text omitted.") from None
    return source, tree, hashlib.sha256(raw).hexdigest()


def fingerprint(node):
    return hashlib.sha256(ast.dump(node, include_attributes=False).encode()).hexdigest()


def targets(node):
    if isinstance(node, ast.Name):
        return [node.id]
    if isinstance(node, (ast.Tuple, ast.List)):
        return [name for child in node.elts for name in targets(child)]
    return []


def describe(node):
    label = type(node).__name__
    if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
        label += " " + node.name
    elif isinstance(node, ast.Assign):
        names = [name for target in node.targets for name in targets(target)]
        label += " " + ", ".join(names[:8])
    elif isinstance(node, ast.Import):
        label += " " + ", ".join(item.name for item in node.names)
    elif isinstance(node, ast.ImportFrom):
        label += " " + (node.module or "")
    return f"lines {node.lineno}-{node.end_lineno}: {label.strip()}"


def extraction_region(source, tree, heading):
    starts = [node for node in tree.body if isinstance(node, ast.Assign)
              and any("API_VERSION" in targets(target) for target in node.targets)]
    endings = [token.start[0] for token in tokenize.generate_tokens(io.StringIO(source).readline)
               if token.type == tokenize.COMMENT and token.start[1] == 0
               and re.fullmatch(r"#\s*=+\s*" + heading + r"\s*=+\s*", token.string)]
    if len(starts) != 1 or len(endings) != 1:
        return None
    start, end = starts[0].lineno, endings[0]
    if end <= start:
        return None
    nodes = [node for node in tree.body if start <= node.lineno < end]
    if any(node.end_lineno >= end for node in nodes):
        return None
    return [fingerprint(node) for node in nodes], start, end - 1


def main():
    root = Path.home() / "Desktop" / "wrike-local-baseline"
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--backup", type=Path, default=root / "Wrike_Data_old_backup.py")
    parser.add_argument("--local", type=Path, default=root / "Wrike_Data_local_validation.py")
    args = parser.parse_args()
    paths = [args.backup.expanduser().resolve(), args.local.expanduser().resolve()]
    if paths[0] == paths[1]:
        raise ValueError("Backup and validation must be different files.")
    loaded = []
    for label, path in zip(("BACKUP", "VALIDATION"), paths):
        try:
            loaded.append(read_source(path))
        except FileNotFoundError:
            raise ValueError(f"{label} file not found: {path}") from None
        print(f"{label}: {path}")
        print(f"SHA256: {loaded[-1][2]}")
    old_source, old_tree, _ = loaded[0]
    new_source, new_tree, _ = loaded[1]
    old_keys = [fingerprint(node) for node in old_tree.body]
    new_keys = [fingerprint(node) for node in new_tree.body]
    matcher = difflib.SequenceMatcher(a=old_keys, b=new_keys, autojunk=False)
    changes = []
    unchanged = 0
    for operation, a, b, c, d in matcher.get_opcodes():
        if operation == "equal":
            unchanged += b - a
        else:
            changes.append((operation, a, b, c, d))
    print(f"\nUnchanged top-level statements: {unchanged}")
    print(f"Changed blocks: {len(changes)}")
    for index, (operation, a, b, c, d) in enumerate(changes, 1):
        print(f"\nBlock {index}: {operation.upper()}")
        for node in old_tree.body[a:b]:
            print("  BACKUP     " + describe(node))
        for node in new_tree.body[c:d]:
            print("  VALIDATION " + describe(node))
    old_region = extraction_region(old_source, old_tree, "SNOWFLAKE UPLOAD")
    new_region = extraction_region(new_source, new_tree, "LOCAL CSV EXPORT")
    print("\nExtraction block comparison (API_VERSION through export heading):")
    if old_region is None or new_region is None:
        print("NOT DETERMINED: unique top-level boundaries were not found.")
    else:
        print(f"  BACKUP lines {old_region[1]}-{old_region[2]}")
        print(f"  VALIDATION lines {new_region[1]}-{new_region[2]}")
        print("  MATCH" if old_region[0] == new_region[0] else "  DIFFERENCES: review changed blocks above.")
    calls = []
    for node in ast.walk(new_tree):
        if isinstance(node, ast.Call):
            name = getattr(node.func, "id", getattr(node.func, "attr", ""))
            if name in {"get_snowflake_connection", "write_pandas"}:
                calls.append((node.lineno, name))
    print(f"\nRemaining get_snowflake_connection/write_pandas calls: {len(calls)}")
    for line, name in sorted(calls):
        print(f"  Line {line}: {name}")
    print("\nRead-only check complete. Neither script was executed or modified.")
    print("Source values and credentials were not printed.")
    print("This checks saved files, not the version already loaded by a running process.")
    print("MATCH applies only to the marked block. Review all other changed blocks.")
    print("The backup's production commit, token identity and output parity remain separate checks.")


if __name__ == "__main__":
    try:
        main()
    except (OSError, ValueError, UnicodeError, LookupError) as error:
        raise SystemExit(f"CHECK STOPPED: {error}")
