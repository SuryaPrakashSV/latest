from pathlib import Path
import pandas as pd


# =========================================================
# FILES
# =========================================================

BASE_DIR = Path(__file__).resolve().parent

MY_FILE = BASE_DIR / "Single_execut_my_token.xlsx"
AKASH_FILE = BASE_DIR / "Single_execut_akash_token.xlsx"

OUTPUT_FILE = BASE_DIR / "token_project_comparison_with_names.xlsx"


# =========================================================
# HELPERS
# =========================================================

def clean(value):
    if pd.isna(value):
        return ""

    value = str(value).strip()

    if value.lower() in {
        "",
        "nan",
        "none",
        "null",
        "<na>",
    }:
        return ""

    return value


def load_projects(path, token_name):

    print(f"\nLoading {path.name} ...")

    df = pd.read_excel(
        path,
        dtype=str,
        engine="openpyxl",
        usecols=lambda c: str(c).strip() in {
            "id",
            "key",
            "title",
            "is_project"
        }
    ).fillna("")

    df.columns = [
        str(c).strip()
        for c in df.columns
    ]

    for col in ["id", "key", "title", "is_project"]:
        df[col] = df[col].map(clean)

    df["is_project"] = (
        df["is_project"]
        .str.upper()
    )

    # -----------------------------------------------------
    # KEEP ONLY THE ACTUAL PROJECT/FOLDER ROW
    #
    # Project row:
    #     id  = project ID
    #     key = project ID
    #     title = project name
    #
    # Task row:
    #     id  = project/folder ID
    #     key = task ID
    # -----------------------------------------------------

    projects = df[
        (df["is_project"] == "Y")
        & (df["id"] != "")
        & (df["id"].str.upper() != "PLACEHOLDER")
        & (df["key"] == df["id"])
        & (df["title"] != "")
    ].copy()

    projects = (
        projects[
            [
                "id",
                "title"
            ]
        ]
        .drop_duplicates()
        .rename(
            columns={
                "id": "project_id",
                "title": "project_name"
            }
        )
    )

    # If same project appears multiple times because of owners,
    # keep one ID/name combination.
    projects = (
        projects
        .drop_duplicates(
            subset=["project_id"]
        )
        .sort_values(
            ["project_name", "project_id"]
        )
        .reset_index(drop=True)
    )

    projects.insert(
        0,
        "token_source",
        token_name
    )

    print(
        f"Unique projects with names: "
        f"{len(projects):,}"
    )

    return projects


# =========================================================
# LOAD BOTH FILES
# =========================================================

my_projects = load_projects(
    MY_FILE,
    "MY_TOKEN"
)

akash_projects = load_projects(
    AKASH_FILE,
    "AKASH_TOKEN"
)


# =========================================================
# PROJECT ID SETS
# =========================================================

my_ids = set(
    my_projects["project_id"]
)

akash_ids = set(
    akash_projects["project_id"]
)

common_ids = my_ids & akash_ids

only_my_ids = my_ids - akash_ids

only_akash_ids = akash_ids - my_ids


# =========================================================
# MY TOKEN ONLY
# =========================================================

only_my = (
    my_projects[
        my_projects["project_id"].isin(
            only_my_ids
        )
    ]
    .copy()
    .reset_index(drop=True)
)


# =========================================================
# AKASH TOKEN ONLY
# =========================================================

only_akash = (
    akash_projects[
        akash_projects["project_id"].isin(
            only_akash_ids
        )
    ]
    .copy()
    .reset_index(drop=True)
)


# =========================================================
# COMMON PROJECTS
# =========================================================

common = (
    my_projects[
        my_projects["project_id"].isin(
            common_ids
        )
    ][
        [
            "project_id",
            "project_name"
        ]
    ]
    .merge(
        akash_projects[
            akash_projects["project_id"].isin(
                common_ids
            )
        ][
            [
                "project_id",
                "project_name"
            ]
        ],
        on="project_id",
        how="inner",
        suffixes=(
            "_my_token",
            "_akash_token"
        )
    )
)

common["project_name_match"] = (
    common["project_name_my_token"]
    == common["project_name_akash_token"]
)


# =========================================================
# SUMMARY
# =========================================================

summary = pd.DataFrame(
    [
        {
            "metric": "My token projects",
            "count": len(my_projects)
        },
        {
            "metric": "Akash token projects",
            "count": len(akash_projects)
        },
        {
            "metric": "Common projects",
            "count": len(common)
        },
        {
            "metric": "Only my token",
            "count": len(only_my)
        },
        {
            "metric": "Only Akash token",
            "count": len(only_akash)
        },
    ]
)


# =========================================================
# TERMINAL RESULTS
# =========================================================

print("\n" + "=" * 70)
print("PROJECT ID + NAME COMPARISON")
print("=" * 70)

print(
    f"My token projects       : "
    f"{len(my_projects):,}"
)

print(
    f"Akash token projects    : "
    f"{len(akash_projects):,}"
)

print(
    f"Common projects         : "
    f"{len(common):,}"
)

print(
    f"Only in my token        : "
    f"{len(only_my):,}"
)

print(
    f"Only in Akash token     : "
    f"{len(only_akash):,}"
)


# =========================================================
# EXCEL OUTPUT
# =========================================================

with pd.ExcelWriter(
    OUTPUT_FILE,
    engine="openpyxl"
) as writer:

    summary.to_excel(
        writer,
        sheet_name="Summary",
        index=False
    )

    my_projects.to_excel(
        writer,
        sheet_name="My_Token_Projects",
        index=False
    )

    akash_projects.to_excel(
        writer,
        sheet_name="Akash_Token_Projects",
        index=False
    )

    common.to_excel(
        writer,
        sheet_name="Common_Projects",
        index=False
    )

    only_my.to_excel(
        writer,
        sheet_name="Only_My_Token",
        index=False
    )

    only_akash.to_excel(
        writer,
        sheet_name="Only_Akash_Token",
        index=False
    )


print("\nComparison complete.")
print(f"Report created: {OUTPUT_FILE}")
