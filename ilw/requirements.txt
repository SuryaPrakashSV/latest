SELECT
    COUNT(DISTINCT "key")                                                   AS total_entities,
    COUNT(DISTINCT CASE WHEN "key" =  "id" THEN "key" END)                  AS folders_and_projects,
    COUNT(DISTINCT CASE WHEN "key" <> "id" THEN "key" END)                  AS tasks_key_ne_id,
    COUNT(DISTINCT CASE WHEN "key" IN ("parent_task_id","child_task_id","grandchild_task_id",
                                       "baby_task_id","grandbaby_task_id","great_grandbaby_task_id")
                        THEN "key" END)                                     AS tasks_hierarchy
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
WHERE NULLIF(TRIM("key"), '') IS NOT NULL
  AND UPPER(TRIM("key")) <> 'PLACEHOLDER';
