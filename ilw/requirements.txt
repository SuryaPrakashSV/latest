SELECT DISTINCT
    "id",
    "key",
    "title",
    "is_project"
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
WHERE "title" ILIKE '%BTS EMEA Takedown%';


SELECT
    TABLE_CATALOG,
    TABLE_SCHEMA,
    TABLE_NAME,
    TABLE_TYPE
FROM GBI_RETAIL_BAP_DB.INFORMATION_SCHEMA.TABLES
WHERE TABLE_NAME ILIKE '%WRIKE%'
ORDER BY TABLE_SCHEMA, TABLE_NAME;
