WITH task_statuses AS (
    SELECT
        "key" AS task_id,
        COALESCE(
            NULLIF(TRIM("status"), ''),
            '[Missing]'
        ) AS general_status,
        COALESCE(
            NULLIF(TRIM("customStatusId"), ''),
            '[Missing or unmapped]'
        ) AS detailed_status
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
    WHERE NULLIF(TRIM("key"), '') IS NOT NULL
      AND UPPER(TRIM("key")) <> 'PLACEHOLDER'
      AND "key" IN (
          "parent_task_id",
          "child_task_id",
          "grandchild_task_id",
          "baby_task_id",
          "grandbaby_task_id",
          "great_grandbaby_task_id"
      )
)
SELECT
    general_status,
    detailed_status,
    COUNT(DISTINCT task_id) AS distinct_task_count
FROM task_statuses
GROUP BY
    general_status,
    detailed_status
ORDER BY
    general_status,
    distinct_task_count DESC,
    detailed_status;



SELECT COUNT(DISTINCT "key") AS distinct_tasks
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"

  AND UPPER(TRIM("key")) <> 'PLACEHOLDER'
  AND "key" IN ("parent_task_id","child_task_id","grandchild_task_id",
                "baby_task_id","grandbaby_task_id","great_grandbaby_task_id");
