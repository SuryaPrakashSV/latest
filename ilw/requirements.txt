"""Quick one-task probe: which of these field names are valid EXPLICIT
optional fields on GET /tasks/{id}?

We already have confirmed, literal proof that 'title' and 'superTaskIds'
are REJECTED when requested explicitly on this endpoint (they come back by
default instead) - see the two 400 errors already hit and fixed earlier in
this project. Before building a full hierarchy diagnostic across all 316
Holiday tasks using parentIds/superParentIds/subTaskIds, check each one
individually against a single real task ID, so we don't repeat that same
trial-and-error cycle at scale.

Usage:
    python3 probe_fields.py
    (it will ask for a task ID interactively)
"""

import getpass
import os

import requests

API_BASE = "https://www.wrike.com/api/v4"


def token() -> str:
    value = os.getenv("WRIKE_TOKEN", "").strip()
    return value or getpass.getpass("Wrike token (hidden; not saved): ").strip()


def main() -> None:
    tok = token()
    headers = {"Authorization": f"Bearer {tok}"}

    sample_task_id = input("Paste one real task ID to probe against: ").strip()

    candidate_fields = ["parentIds", "superParentIds", "subTaskIds", "superTaskIds", "title"]

    print(f"\nProbing {sample_task_id} field-by-field...\n")
    for field in candidate_fields:
        resp = requests.get(
            f"{API_BASE}/tasks/{sample_task_id}",
            headers=headers,
            params={"fields": f'["{field}"]'},
        )
        if resp.ok:
            data = resp.json()["data"][0]
            present_keys = sorted(data.keys())
            print(f"{field}: OK (accepted). Full response keys: {present_keys}")
        else:
            print(f"{field}: REJECTED - {resp.text}")


if __name__ == "__main__":
    main()
