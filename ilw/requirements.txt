WITH holiday_rows AS (
    SELECT *
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
    WHERE "id" = 'MQAAAAEIsSOV'
),
all_hierarchy_ids AS (
    SELECT NULLIF(TRIM("parent_task_id"), '') AS task_id FROM holiday_rows
    UNION ALL
    SELECT NULLIF(TRIM("child_task_id"), '') FROM holiday_rows
    UNION ALL
    SELECT NULLIF(TRIM("grandchild_task_id"), '') FROM holiday_rows
    UNION ALL
    SELECT NULLIF(TRIM("baby_task_id"), '') FROM holiday_rows
    UNION ALL
    SELECT NULLIF(TRIM("grandbaby_task_id"), '') FROM holiday_rows
    UNION ALL
    SELECT NULLIF(TRIM("great_grandbaby_task_id"), '') FROM holiday_rows
)
SELECT DISTINCT task_id
FROM all_hierarchy_ids
WHERE task_id IS NOT NULL
ORDER BY task_id;
