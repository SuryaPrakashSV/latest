#!/usr/bin/env python3
"""Replace the known final Snowflake upload block in the Mac validation copy.

This script never imports or runs the extraction. It reads and checks the source,
backs up the original bytes, then replaces only the final upload section with a
local CSV export. Unexpected source structure stops the repair without edits.

Run: python3 repair_local_csv_export.py
"""

import argparse
import ast
import hashlib
import os
import re
import tempfile
from datetime import datetime, timezone
from pathlib import Path


EXPORT = '''# ==================== LOCAL CSV EXPORT ====================
from pathlib import Path
from datetime import datetime, timezone

if df_final_complete.empty:
    raise RuntimeError("No rows extracted. Cannot validate this run.")

run_id = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%S_%fZ")
output_dir = (
    Path(__file__).resolve().parent
    / "output"
    / "local_validation"
    / run_id
)
output_dir.mkdir(parents=True, exist_ok=False)
output_file = output_dir / "wrike_local_full.csv"
temporary_output_file = output_dir / "wrike_local_full.csv.partial"

df_final_complete.to_csv(
    temporary_output_file,
    index=False,
    encoding="utf-8-sig",
)
temporary_output_file.replace(output_file)

print(f"Saved: {output_file}", flush=True)
print(f"Dataset rows exported: {len(df_final_complete):,}", flush=True)
'''


def called_name(node):
    return getattr(node.func, "id", getattr(node.func, "attr", ""))


def remaining_upload_calls(tree):
    return [node for node in ast.walk(tree) if isinstance(node, ast.Call)
            and called_name(node) in {"get_snowflake_connection", "write_pandas"}]


def build_replacement(source):
    tree = ast.parse(source)
    markers = list(re.finditer(r"(?m)^#[^\r\n]*SNOWFLAKE UPLOAD[^\r\n]*$", source))
    if len(markers) != 1:
        raise ValueError("Expected exactly one top-level SNOWFLAKE UPLOAD heading; found " + str(len(markers)))
    if "LOCAL CSV EXPORT" in source:
        raise ValueError("An existing LOCAL CSV EXPORT section was found. Review its location before repairing.")
    marker = markers[0]
    prefix = source[:marker.start()]
    first_line = source.count("\n", 0, marker.start()) + 1
    calls = remaining_upload_calls(tree)
    if sorted(called_name(node) for node in calls) != ["get_snowflake_connection", "write_pandas"]:
        raise ValueError("Expected one call to each Snowflake upload function; source differs from the reviewed block.")
    if any(node.lineno < first_line for node in calls):
        raise ValueError("A Snowflake call exists before the final upload section. No changes made.")

    tail = ast.parse(source[marker.start():])
    if sum(isinstance(node, ast.Try) for node in tail.body) != 1:
        raise ValueError("The final upload section does not have the expected top-level try block.")
    for node in tail.body:
        if isinstance(node, ast.Try):
            continue
        if (isinstance(node, ast.Expr) and isinstance(node.value, ast.Call)
                and isinstance(node.value.func, ast.Name) and node.value.func.id == "print"):
            continue
        if (isinstance(node, ast.Assign) and len(node.targets) == 1
                and isinstance(node.targets[0], ast.Name) and node.targets[0].id == "connection"
                and isinstance(node.value, ast.Constant) and node.value.value is None):
            continue
        raise ValueError("Unexpected code after the upload heading. No changes made.")

    allowed_calls = {"get_snowflake_connection", "write_pandas", "print", "len",
                     "ValueError", "print_exc", "exit", "close"}
    allowed_targets = {"connection", "table_name", "success", "nchunks", "nrows", "_"}
    for node in ast.walk(tail):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef, ast.Import,
                             ast.ImportFrom, ast.For, ast.While, ast.With, ast.AugAssign, ast.AnnAssign)):
            raise ValueError("Unexpected code structure in the upload section. No changes made.")
        if isinstance(node, ast.Call) and called_name(node) not in allowed_calls:
            raise ValueError("Unexpected call in the upload section: " + called_name(node))
        if isinstance(node, ast.Assign):
            for target in node.targets:
                targets = target.elts if isinstance(target, (ast.Tuple, ast.List)) else [target]
                if any(not isinstance(item, ast.Name) or item.id not in allowed_targets for item in targets):
                    raise ValueError("Unexpected assignment in the upload section. No changes made.")

    result = prefix + EXPORT
    checked = ast.parse(result)
    compile(result, "Wrike_Data_local_validation.py", "exec")  # Compile only. Never execute.
    if remaining_upload_calls(checked):
        raise ValueError("A Snowflake upload call remains. No changes made.")
    prefix_tree = ast.parse(prefix)
    kept_tree = ast.Module(body=checked.body[:len(prefix_tree.body)], type_ignores=[])
    if ast.dump(prefix_tree) != ast.dump(kept_tree):
        raise ValueError("Code before the upload section changed. No changes made.")
    return result, first_line


def repair(path):
    path = path.expanduser().resolve()
    if path.name != "Wrike_Data_local_validation.py":
        raise ValueError("Only Wrike_Data_local_validation.py can be repaired by this utility.")
    original = path.read_bytes()
    changed, first_line = build_replacement(original.decode("utf-8-sig"))
    replacement = changed.encode("utf-8")
    if original.startswith(b"\xef\xbb\xbf"):
        replacement = b"\xef\xbb\xbf" + replacement
    run_id = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%S_%fZ")
    backup = path.with_name(path.name + ".before_csv_repair_" + run_id + ".bak")
    if path.read_bytes() != original:
        raise ValueError("Source changed during inspection. Save it and try again.")
    with backup.open("xb") as handle:
        handle.write(original)
    temp_name = None
    try:
        with tempfile.NamedTemporaryFile(dir=path.parent, prefix=".csv_repair_", suffix=".tmp", delete=False) as handle:
            temp_name = handle.name
            handle.write(replacement)
            handle.flush()
            os.fsync(handle.fileno())
        os.chmod(temp_name, path.stat().st_mode & 0o777)
        if path.read_bytes() != original:
            raise ValueError("Source changed before replacement. Original remains untouched.")
        os.replace(temp_name, path)
        temp_name = None
    finally:
        if temp_name is not None:
            Path(temp_name).unlink(missing_ok=True)
    if path.read_bytes() != replacement:
        raise ValueError("Saved file verification failed. Backup: " + str(backup))
    print("REPAIR COMPLETE")
    print("Validation file: " + str(path))
    print("Backup: " + str(backup))
    print("Replaced final upload section starting at old line: " + str(first_line))
    print("Code before that section: unchanged")
    print("Remaining get_snowflake_connection/write_pandas calls: 0")
    print("Syntax check: passed")
    print("Local CSV export: installed; incomplete writes retain a .partial extension")
    print("Source SHA256: " + hashlib.sha256(replacement).hexdigest())
    print("The extraction has NOT been run. No network requests were made.")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--file", type=Path, default=Path.home() / "Desktop" / "wrike-local-baseline" / "Wrike_Data_local_validation.py")
    args = parser.parse_args()
    try:
        repair(args.file)
    except (OSError, ValueError, SyntaxError, UnicodeError) as exc:
        raise SystemExit("STOP: " + str(exc))
