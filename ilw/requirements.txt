WITH task_rows AS (
    SELECT
        "status" AS task_status,
        "customStatusId" AS custom_status,
        "updatedDate" AS updated_date,
        "completedDate" AS completed_date
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
    WHERE "parent_task_id" = 'IEADQP2VKRD2ARI6'
       OR "child_task_id" = 'IEADQP2VKRD2ARI6'
       OR "grandchild_task_id" = 'IEADQP2VKRD2ARI6'
       OR "baby_task_id" = 'IEADQP2VKRD2ARI6'
       OR "grandbaby_task_id" = 'IEADQP2VKRD2ARI6'
       OR "great_grandbaby_task_id" = 'IEADQP2VKRD2ARI6'
),
stored_statuses AS (
    SELECT
        task_status,
        custom_status,
        updated_date,
        completed_date,
        COUNT(*) AS physical_row_count
    FROM task_rows
    GROUP BY
        task_status,
        custom_status,
        updated_date,
        completed_date
)
SELECT
    ROW_NUMBER() OVER (
        ORDER BY updated_date
    ) AS recorded_sequence,
    task_status,
    custom_status,
    updated_date,
    completed_date,
    physical_row_count
FROM stored_statuses
ORDER BY recorded_sequence;
