SELECT COLUMN_NAME, DATA_TYPE, ORDINAL_POSITION
FROM GBI_RETAIL_BAP_DB.INFORMATION_SCHEMA.COLUMNS
WHERE UPPER(TABLE_SCHEMA) = 'ASO_OPS_WSA_SNDBX_BIZ_APP'
  AND UPPER(TABLE_NAME)   = 'WRIKE_STG'
  AND (LOWER(COLUMN_NAME) LIKE '%creat%'
    OR LOWER(COLUMN_NAME) LIKE '%start%'
    OR LOWER(COLUMN_NAME) LIKE '%date%')
ORDER BY ORDINAL_POSITION;
