WITH source_data AS (
    SELECT
        NULLIF(TRIM("id"), '')         AS container_id,
        NULLIF(TRIM("key"), '')        AS entity_id,
        UPPER(TRIM("is_project"))      AS project_flag
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
),
folder_keys AS (
    SELECT DISTINCT entity_id
    FROM source_data
    WHERE entity_id = container_id
      AND entity_id IS NOT NULL
      AND UPPER(entity_id) <> 'PLACEHOLDER'
),
container_counts AS (
    SELECT
        COUNT(DISTINCT container_id)
            AS all_container_ids_including_folders,

        COUNT(DISTINCT CASE
            WHEN entity_id = container_id
             AND project_flag = 'Y'
            THEN container_id
        END) AS confirmed_project_ids
    FROM source_data
    WHERE container_id IS NOT NULL
      AND UPPER(container_id) <> 'PLACEHOLDER'
),
task_counts AS (
    SELECT
        COUNT(DISTINCT s.entity_id) AS task_ids_using_your_key_rule
    FROM source_data s
    WHERE s.entity_id IS NOT NULL
      AND UPPER(s.entity_id) <> 'PLACEHOLDER'
      AND s.entity_id <> s.container_id
      AND NOT EXISTS (
          SELECT 1
          FROM folder_keys f
          WHERE f.entity_id = s.entity_id
      )
)
SELECT
    c.all_container_ids_including_folders,
    c.confirmed_project_ids,
    t.task_ids_using_your_key_rule
FROM container_counts c
CROSS JOIN task_counts t;




--======================

SELECT
    TRIM("id") AS container_id,

    MAX(CASE
        WHEN TRIM("key") = TRIM("id")
        THEN NULLIF(TRIM("title"), '')
    END) AS container_name,

    CASE
        WHEN MAX(CASE
            WHEN TRIM("key") = TRIM("id")
             AND UPPER(TRIM("is_project")) = 'Y'
            THEN 1 ELSE 0
        END) = 1
        THEN 'CONFIRMED_PROJECT'
        ELSE 'NOT_CONFIRMED_AS_PROJECT'
    END AS classification

FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
WHERE NULLIF(TRIM("id"), '') IS NOT NULL
  AND UPPER(TRIM("id")) <> 'PLACEHOLDER'
GROUP BY TRIM("id")
ORDER BY container_name, container_id;
