"""Check whether tasks missing from Wrike_STG cluster at a particular
hierarchy depth (subtasks of subtasks) - the leading hypothesis for why a
small project (11/11 matched) reconciled perfectly while a large project
(316 API tasks, only 193 in Snowflake) shows a 123-task gap.

For each missing task, this looks at its immediate `superTaskIds` (its
direct parent task, if any - not the full depth to the project root, just
whether it's nested at all vs. sitting directly under the project folder).
If the 123 missing tasks are overwhelmingly "has a superTaskId" while the
193 tasks that DID make it into Snowflake are overwhelmingly top-level,
that's strong evidence the production extraction's hierarchy walk is
dropping deeper subtasks - the same class of bug we hit ourselves earlier
in this project before we added the closure-walk over subTaskIds.

Usage:
    python3 inspect_missing_tasks.py \
        --missing-csv output/task_reconciliation/MQAAAAEIsSOV_api_only_missing_from_snowflake.csv
"""

from __future__ import annotations

import argparse
import csv
import getpass
import os

import requests

API_BASE = "https://www.wrike.com/api/v4"


def token() -> str:
    value = os.getenv("WRIKE_TOKEN", "").strip()
    return value or getpass.getpass("Wrike token (hidden; not saved): ").strip()


def chunked(items, size):
    items = list(items)
    for i in range(0, len(items), size):
        yield items[i : i + size]


def read_task_ids(csv_path: str) -> list[str]:
    with open(csv_path, newline="", encoding="utf-8-sig") as f:
        reader = csv.DictReader(f)
        fieldnames = {(name or "").strip().lower(): name for name in (reader.fieldnames or [])}
        key = fieldnames.get("task_id")
        if key is None:
            raise SystemExit(f"No task_id column found in {csv_path}. Columns: {reader.fieldnames}")
        return [row[key].strip() for row in reader if row.get(key, "").strip()]


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--missing-csv", required=True)
    args = parser.parse_args()

    missing_ids = read_task_ids(args.missing_csv)
    tok = token()
    headers = {"Authorization": f"Bearer {tok}"}

    has_super = 0
    no_super = 0
    parent_count_dist: dict[int, int] = {}
    sample_top_level_titles: list[str] = []

    for batch in chunked(missing_ids, 100):
        ids_param = ",".join(batch)
        resp = requests.get(
            f"{API_BASE}/tasks/{ids_param}",
            headers=headers,
            params={"fields": '["superTaskIds","title"]'},
        )
        resp.raise_for_status()
        for task in resp.json()["data"]:
            supers = task.get("superTaskIds") or []
            parent_count_dist[len(supers)] = parent_count_dist.get(len(supers), 0) + 1
            if supers:
                has_super += 1
            else:
                no_super += 1
                if len(sample_top_level_titles) < 10:
                    sample_top_level_titles.append(task.get("title", "<no title>"))

    total = len(missing_ids)
    print(f"Missing tasks checked: {total}")
    print(f"Has >=1 superTaskId (nested under another task): {has_super} ({has_super / total:.0%})")
    print(f"No superTaskIds (sits directly under the project): {no_super} ({no_super / total:.0%})")
    print("\nImmediate-parent-count distribution:")
    for count, num in sorted(parent_count_dist.items()):
        print(f"  {count} parent(s): {num} tasks")

    if sample_top_level_titles:
        print("\nSample titles of the top-level (no superTaskId) missing tasks:")
        for title in sample_top_level_titles:
            print(f"  - {title}")


if __name__ == "__main__":
    main()
