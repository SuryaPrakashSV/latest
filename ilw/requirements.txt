SELECT
    "id" AS project_id,
    MAX(TRIM("title")) AS project_name
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
WHERE "key" = "id"
  AND NULLIF(TRIM("id"), '') IS NOT NULL
  AND UPPER(TRIM("id")) <> 'PLACEHOLDER'
  AND NULLIF(TRIM("title"), '') IS NOT NULL
GROUP BY "id"
ORDER BY project_name, project_id;
