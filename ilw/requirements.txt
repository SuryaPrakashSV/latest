Step 1 — check if "Specialist image refresh Q1&Q2 FY27" has its own record in Wrike_STG:

SELECT DISTINCT
    "id" AS project_or_folder_id,
    "title" AS project_or_folder_name,
    "is_project"
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
WHERE "id" = 'MQAAAAE06vxh'
   OR UPPER("title") LIKE '%SPECIALIST IMAGE REFRESH Q1&Q2 FY27%'
ORDER BY project_or_folder_name;

Step 2 — if that returns a row, combine Holiday + Specialist and rerun the full-hierarchy extraction across both:

WITH target_rows AS (
    SELECT *
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
    WHERE "id" IN ('MQAAAAEIsSOV', 'MQAAAAE06vxh')
),
all_hierarchy_ids AS (
    SELECT NULLIF(TRIM("parent_task_id"), '') AS task_id FROM target_rows
    UNION ALL
    SELECT NULLIF(TRIM("child_task_id"), '') FROM target_rows
    UNION ALL
    SELECT NULLIF(TRIM("grandchild_task_id"), '') FROM target_rows
    UNION ALL
    SELECT NULLIF(TRIM("baby_task_id"), '') FROM target_rows
    UNION ALL
    SELECT NULLIF(TRIM("grandbaby_task_id"), '') FROM target_rows
    UNION ALL
    SELECT NULLIF(TRIM("great_grandbaby_task_id"), '') FROM target_rows
)
SELECT DISTINCT task_id AS "task_id"
FROM all_hierarchy_ids
WHERE task_id IS NOT NULL
ORDER BY "task_id";
