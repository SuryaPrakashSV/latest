"""End-to-end project task reconciliation, Wrike API vs Snowflake, with NO
manual CSV export/download/upload step anywhere in the path.

This replaces the earlier reconsile_task_ids.py + manual-Snowsight-query +
CSV-download workflow, which caused several real, silent bugs this session
(BOM characters, TASK_ID/task_id casing mismatches, an empty uploaded
table, and a false "still missing" result for Holiday that was only caught
by rerunning through a direct Snowflake connection instead). Everything
here goes straight from the Wrike API and Snowflake into Python and back,
so there's no manual re-typing or file round-tripping to silently corrupt.

For a project's full picture, it:
  1. Pulls the live task count from Wrike's own authoritative
     GET /folders/{id}/tasks?descendants=true&subTasks=true call (paginated).
  2. Queries Snowflake directly for that project's task_id set, using the
     ALL-hierarchy-column union (not just the deepest value - COALESCE
     undercounting is a known, confirmed issue for some projects).
  3. Compares them: matched / API-only (missing under this project's own
     Snowflake rows) / Snowflake-only (orphans).
  4. For any API-only tasks, immediately checks the ENTIRE Wrike_STG table
     (no project filter) to see if they're actually attributed to a
     DIFFERENT project - distinguishing "real gap" from "attribution-scope
     difference" in one run, instead of a separate manual follow-up step.

Usage:
    python3 reconcile_project_direct.py \
        --project-id MQAAAABnlKFK \
        --project-name "ASA OS Compatibility"
"""

from __future__ import annotations

import argparse
import getpass
import os
import time
from datetime import datetime, timezone
from email.utils import parsedate_to_datetime

import requests

API_BASE = "https://www.wrike.com/api/v4"
TABLE_FQN = 'GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"'

HIERARCHY_COLUMNS = [
    "parent_task_id",
    "child_task_id",
    "grandchild_task_id",
    "baby_task_id",
    "grandbaby_task_id",
    "great_grandbaby_task_id",
]


def wrike_token() -> str:
    value = os.getenv("WRIKE_TOKEN", "").strip()
    return value or getpass.getpass("Wrike token (hidden; not saved): ").strip()


def get_with_backoff(url, headers, params=None, max_retries=6, timeout=60):
    for attempt in range(max_retries):
        resp = requests.get(url, headers=headers, params=params, timeout=timeout)
        if resp.status_code != 429 or attempt == max_retries - 1:
            return resp
        retry_after = resp.headers.get("Retry-After")
        wait = None
        if retry_after:
            try:
                wait = max(0.0, float(retry_after))
            except ValueError:
                try:
                    retry_dt = parsedate_to_datetime(retry_after)
                    if retry_dt.tzinfo is None:
                        retry_dt = retry_dt.replace(tzinfo=timezone.utc)
                    wait = max(0.0, (retry_dt - datetime.now(timezone.utc)).total_seconds())
                except (TypeError, ValueError):
                    wait = None
        if wait is None:
            wait = min(120, 5 * (2 ** attempt))
        print(f"Rate limited (429). Waiting {wait:.0f}s before retry {attempt + 1}/{max_retries}...")
        time.sleep(wait)
    return resp


def fetch_all_folder_tasks(project_id: str, headers: dict) -> set[str]:
    all_ids: set[str] = set()
    next_page_token = None
    page = 1
    while True:
        params = (
            {"nextPageToken": next_page_token}
            if next_page_token
            else {"descendants": "true", "subTasks": "true", "fields": '["subTaskIds"]'}
        )
        resp = get_with_backoff(f"{API_BASE}/folders/{project_id}/tasks", headers, params=params)
        resp.raise_for_status()
        payload = resp.json()
        page_tasks = payload.get("data", [])
        all_ids.update(t["id"] for t in page_tasks)
        print(f"  Page {page}: {len(page_tasks)} tasks (running total {len(all_ids)})")
        next_page_token = payload.get("nextPageToken")
        if not next_page_token:
            break
        page += 1
    return all_ids


def snowflake_connection_from_env():
    import snowflake.connector

    authenticator = os.getenv("SNOWFLAKE_AUTHENTICATOR", "snowflake").strip().lower()
    base_required = ["SNOWFLAKE_ACCOUNT", "SNOWFLAKE_USER", "SNOWFLAKE_WAREHOUSE"]
    required = base_required if authenticator == "externalbrowser" else base_required + ["SNOWFLAKE_PASSWORD"]
    missing = [name for name in required if not os.getenv(name)]
    if missing:
        raise SystemExit(f"Missing required environment variable(s) for Snowflake: {missing}.")

    kwargs = dict(
        account=os.environ["SNOWFLAKE_ACCOUNT"],
        user=os.environ["SNOWFLAKE_USER"],
        warehouse=os.environ["SNOWFLAKE_WAREHOUSE"],
        role=os.getenv("SNOWFLAKE_ROLE"),
        database="GBI_RETAIL_BAP_DB",
        schema="ASO_OPS_WSA_SNDBX_BIZ_APP",
    )
    if authenticator == "externalbrowser":
        kwargs["authenticator"] = "externalbrowser"
        print("Opening a browser window to complete SSO login - finish logging in there, then return here.")
    else:
        kwargs["password"] = os.environ["SNOWFLAKE_PASSWORD"]

    return snowflake.connector.connect(**kwargs)


def snowflake_task_ids_for_project(cursor, project_id: str) -> set[str]:
    union_parts = [
        f'SELECT NULLIF(TRIM("{col}"), \'\') AS task_id FROM {TABLE_FQN} WHERE "id" = %(project_id)s'
        for col in HIERARCHY_COLUMNS
    ]
    query = "SELECT DISTINCT task_id FROM (" + " UNION ALL ".join(union_parts) + ") WHERE task_id IS NOT NULL"
    cursor.execute(query, {"project_id": project_id})
    return {row[0] for row in cursor.fetchall()}


def find_elsewhere(cursor, task_ids: set[str]) -> dict[str, tuple[str, str]]:
    """For each task_id, find which project's rows it actually appears under
    anywhere in the whole table. Returns {task_id: (owning_id, owning_title)}
    for ones found; omits ones that are genuinely absent table-wide.
    """
    if not task_ids:
        return {}

    cursor.execute("CREATE OR REPLACE TEMPORARY TABLE TEMP_RECONCILE_LOOKUP (task_id VARCHAR)")
    cursor.executemany(
        "INSERT INTO TEMP_RECONCILE_LOOKUP (task_id) VALUES (%(task_id)s)",
        [{"task_id": t} for t in task_ids],
    )

    union_parts = [
        f'SELECT "id" AS owning_id, "title" AS owning_title, NULLIF(TRIM("{col}"), \'\') AS task_id FROM {TABLE_FQN}'
        for col in HIERARCHY_COLUMNS
    ]
    query = f"""
        WITH all_hierarchy AS ({" UNION ALL ".join(union_parts)})
        SELECT DISTINCT m.task_id, a.owning_id, a.owning_title
        FROM TEMP_RECONCILE_LOOKUP m
        LEFT JOIN all_hierarchy a ON a.task_id = m.task_id
        WHERE a.owning_id IS NOT NULL
    """
    cursor.execute(query)
    return {row[0]: (row[1], row[2]) for row in cursor.fetchall()}


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--project-id", required=True)
    parser.add_argument("--project-name", default="")
    args = parser.parse_args()

    headers = {"Authorization": f"Bearer {wrike_token()}"}

    print(f"\n{'=' * 70}\nPulling live Wrike task IDs for {args.project_name or args.project_id}\n{'=' * 70}")
    api_ids = fetch_all_folder_tasks(args.project_id, headers)
    print(f"Wrike API total: {len(api_ids)}")

    connection = snowflake_connection_from_env()
    try:
        with connection.cursor() as cursor:
            print(f"\n{'=' * 70}\nQuerying Snowflake directly (all-hierarchy-column union)\n{'=' * 70}")
            snowflake_ids = snowflake_task_ids_for_project(cursor, args.project_id)
            print(f"Snowflake (this project's own rows): {len(snowflake_ids)}")

            matched = api_ids & snowflake_ids
            api_only = api_ids - snowflake_ids
            snowflake_only = snowflake_ids - api_ids

            print(f"\nMatched:                 {len(matched)}")
            print(f"API-only (not under this project's Snowflake rows): {len(api_only)}")
            print(f"Snowflake-only (orphans): {len(snowflake_only)}")

            if api_only:
                print(f"\n{'=' * 70}\nChecking whether the {len(api_only)} API-only tasks exist elsewhere in Wrike_STG\n{'=' * 70}")
                found = find_elsewhere(cursor, api_only)
                truly_absent = api_only - set(found.keys())

                by_project: dict[str, int] = {}
                for owning_id, owning_title in found.values():
                    by_project[f"{owning_id} ({owning_title})"] = by_project.get(f"{owning_id} ({owning_title})", 0) + 1

                print(f"Found under a DIFFERENT project: {len(found)}")
                for label, count in sorted(by_project.items(), key=lambda kv: -kv[1]):
                    print(f"  {count} task(s) -> {label}")
                print(f"Genuinely absent table-wide (real gap): {len(truly_absent)}")
                if truly_absent:
                    print(f"  {sorted(truly_absent)[:20]}{' ...' if len(truly_absent) > 20 else ''}")
    finally:
        connection.close()


if __name__ == "__main__":
    main()
