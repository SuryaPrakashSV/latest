"""Canonical Wrike project-status history transformations.

This module is deliberately side-effect free. It normalizes events from Wrike
webhooks, Audit Log, or BI Export, calculates lifecycle durations, and enriches
only project-level Wrike_STG rows while preserving the input row count.
"""

def _utc(value: Any) -> pd.Timestamp:
    return pd.to_datetime(value, utc=True, errors="coerce")


def _first(mapping: Mapping[str, Any], *keys: str) -> Any:
    for key in keys:
        value = mapping.get(key)
        if value not in (None, ""):
            return value
    return None


def _stable_event_id(row: Mapping[str, Any]) -> str:
    parts = [
        str(row.get("source") or ""),
        str(row.get("project_id") or ""),
        str(row.get("status_changed_at") or ""),
        str(row.get("old_status_id") or row.get("old_status_name") or ""),
        str(row.get("status_id") or row.get("status_name") or ""),
        str(row.get("event_author_id") or ""),
    ]
    return hashlib.sha256("|".join(parts).encode("utf-8")).hexdigest()


def normalize_events(
    events: Iterable[Mapping[str, Any]],
    *,
    source: str,
    project_names: Mapping[str, str] | None = None,
    status_names: Mapping[str, str] | None = None,
) -> pd.DataFrame:
    """Normalize webhook-like or Audit Log-like rows into one stable schema."""
    project_names = project_names or {}
    status_names = status_names or {}
    rows: list[dict[str, Any]] = []
    ingested_at = pd.Timestamp.now(tz="UTC")

    for raw in events:
        event_type = _first(raw, "eventType", "event_type", "operation", "action")
        if event_type and "status" not in str(event_type).lower():
            continue

        details = raw.get("details") if isinstance(raw.get("details"), Mapping) else {}
        project_id = str(
            _first(raw, "projectId", "project_id", "taskId", "objectId", "object_id")
            or _first(details, "projectId", "taskId", "objectId")
            or ""
        ).strip()
        if not project_id:
            continue

        old_status_id = _first(raw, "oldCustomStatusId", "old_status_id") or _first(
            details, "oldCustomStatusId", "oldStatusId", "oldValue"
        )
        status_id = _first(raw, "customStatusId", "status_id") or _first(
            details, "customStatusId", "statusId", "newValue"
        )
        old_status_name = _first(raw, "oldStatus", "old_status_name") or _first(
            details, "oldStatus", "oldStatusName"
        )
        status_name = _first(raw, "status", "status_name") or _first(
            details, "status", "statusName", "newStatus"
        )
        if old_status_name is None and old_status_id is not None:
            old_status_name = status_names.get(str(old_status_id))
        if status_name is None and status_id is not None:
            status_name = status_names.get(str(status_id))

        changed_at = _utc(
            _first(raw, "lastUpdatedDate", "eventDate", "status_changed_at", "timestamp", "createdDate")
            or _first(details, "eventDate", "timestamp", "createdDate")
        )
        row = {
            "project_id": project_id,
            "project_name": _first(raw, "projectName", "project_name") or project_names.get(project_id),
            "old_status_id": old_status_id,
            "old_status_name": old_status_name,
            "status_id": status_id,
            "status_name": status_name,
            "status_changed_at": changed_at,
            "event_author_id": _first(raw, "eventAuthorId", "event_author_id", "userId", "authorId"),
            "source": source.upper(),
            "source_event_id": _first(raw, "sourceEventId", "source_event_id", "idempotencyKey", "id"),
            "source_payload": json.dumps(raw, separators=(",", ":"), default=str),
            "ingested_at": ingested_at,
        }
        row["source_event_id"] = str(row["source_event_id"] or _stable_event_id(row))
        rows.append(row)

    return finalize_history(pd.DataFrame(rows))


def finalize_history(frame: pd.DataFrame) -> pd.DataFrame:
    """Validate, deduplicate, order, and sequence canonical history rows."""
    if frame.empty:
        return pd.DataFrame(columns=HISTORY_COLUMNS)
    required = {"project_id", "status_changed_at", "source_event_id", "source"}
    missing = required.difference(frame.columns)
    if missing:
        raise ValueError(f"History rows are missing columns: {sorted(missing)}")
    output = frame.copy()
    for column in HISTORY_COLUMNS:
        if column not in output.columns:
            output[column] = pd.NA
    output["project_id"] = output["project_id"].astype(str)
    output["status_changed_at"] = pd.to_datetime(output["status_changed_at"], utc=True, errors="coerce")
    output["ingested_at"] = pd.to_datetime(output["ingested_at"], utc=True, errors="coerce")
    output = output.drop_duplicates(subset=["source", "source_event_id"], keep="last")
    output = output.sort_values(
        ["project_id", "status_changed_at", "source_event_id"], na_position="last"
    ).reset_index(drop=True)
    output["status_changed_date"] = output["status_changed_at"].dt.date
    output["status_changed_year"] = output["status_changed_at"].dt.year.astype("Int64")
    output["event_sequence"] = output.groupby("project_id").cumcount() + 1
    return output[HISTORY_COLUMNS]


def trace_project_status_history(
    project: Mapping[str, Any],
    *,
    status_names: Mapping[str, str],
    raw_audit_events: list[Mapping[str, Any]] | None,
    blocked_reason: str | None = None,
) -> dict[str, Any]:
    """Trace one project's full status trail: current state (live) back to its first stage.

    Every status name and date here is either read directly off `project` /
    `status_names` (both fetched from Wrike moments before calling this) or
    normalized from `raw_audit_events` (Wrike's own Audit Log) - nothing is
    authored locally, unlike the old control_result() helper this replaces.
    When `raw_audit_events` is None, `blocked_reason` must explain what
    blocked retrieval (WrikeClient.get() already appends the relevant
    developers.wrike.com doc link to that message) and this returns a
    blocked report instead of inventing any stage to fill the gap.
    """
    project_id = str(project.get("id") or "")
    project_name = project.get("title")
    nested_project = (
        project.get("project") if isinstance(project.get("project"), Mapping) else {}
    )
    current_status_id = project.get("customStatusId") or nested_project.get("customStatusId")
    current_status_name = (
        project.get("status")
        or nested_project.get("status")
        or status_names.get(str(current_status_id or ""))
        or current_status_id
    )

    # completedDate is the ONLY project field Wrike ties to an actual status
    # transition (see FIELD_MEANINGS in wrike_status_api.py) - every other
    # field is current-state or a PM-set planned date, not an event. This is
    # the one status/date pair extractable with zero Audit Log access.
    completed_at = project.get("completedDate") or nested_project.get("completedDate")
    known_facts = (
        [
            {
                "status_name": "Completed",
                "changed_at": completed_at,
                "source": "CURRENT_PROJECT_METADATA (completedDate)",
            }
        ]
        if completed_at
        else []
    )

    if raw_audit_events is None:
        return {
            "project_id": project_id,
            "project_name": project_name,
            "current_status_name": current_status_name,
            "known_facts": known_facts,
            "stages": pd.DataFrame(columns=HISTORY_COLUMNS),
            "blocked": True,
            "blocked_reason": blocked_reason,
            "traces_back_to_current": False,
        }

    stages = normalize_events(
        raw_audit_events,
        source="AUDIT_LOG",
        project_names={project_id: project_name},
        status_names=status_names,
    )
    project_stages = (
        stages[stages["project_id"].astype(str).eq(project_id)].reset_index(drop=True)
        if not stages.empty
        else stages
    )

    reaches_current = (
        not project_stages.empty
        and str(project_stages.iloc[-1]["status_name"]).strip().casefold()
        == str(current_status_name or "").strip().casefold()
    )

    return {
        "project_id": project_id,
        "project_name": project_name,
        "current_status_name": current_status_name,
        "known_facts": known_facts,
        "stages": project_stages,
        "blocked": False,
        "blocked_reason": None,
        "traces_back_to_current": reaches_current,
    }

def build_task_inventory(
    *,
    project_id: str,
    project_name: str | None,
    tasks_by_id: Mapping[str, Mapping[str, Any]],
    status_names: Mapping[str, str],
    contact_names: Mapping[str, str] | None = None,
    status_workflows: Mapping[str, Mapping[str, Any]] | None = None,
) -> pd.DataFrame:
    """Flatten every task/subtask under one project into one inventory row per task.

    Every value here is read directly off Wrike's own Task objects
    (parentIds/superTaskIds/subTaskIds/customStatusId/etc. from
    GET /folders/{id}/tasks and GET /tasks/{ids}) - no nesting depth is
    assumed or hardcoded. A task is classified TASK vs SUBTASK purely from
    whether it has superTaskIds, however many levels deep that chain goes;
    `tasks_by_id` is expected to already be the closure of the whole tree
    (every subTaskIds reference resolved), not just one page of results.

    `status_workflows` maps customStatusId -> {"workflow_id", "workflow_name"}
    (build it from GET /workflows via wrike_status_api.flatten_workflows) so
    the workflow a task's status belongs to is on the row, not just the
    status name - a task with no resolvable status naturally gets blanks
    here rather than a guessed workflow.
    """
    contact_names = contact_names or {}
    status_workflows = status_workflows or {}
    rows: list[dict[str, Any]] = []
    for task_id, task in tasks_by_id.items():
        super_task_ids = [str(i) for i in (task.get("superTaskIds") or [])]
        sub_task_ids = [str(i) for i in (task.get("subTaskIds") or [])]
        parent_folder_ids = [str(i) for i in (task.get("parentIds") or [])]
        responsible_ids = [str(i) for i in (task.get("responsibleIds") or [])]
        author_ids = [str(i) for i in (task.get("authorIds") or [])]
        dates = task.get("dates") if isinstance(task.get("dates"), Mapping) else {}
        status_id = task.get("customStatusId")
        workflow_info = status_workflows.get(str(status_id or ""), {})

        rows.append(
            {
                "project_id": project_id,
                "project_name": project_name,
                "task_id": str(task_id),
                "task_name": task.get("title"),
                "item_type": "SUBTASK" if super_task_ids else "TASK",
                "parent_task_id": super_task_ids[0] if super_task_ids else None,
                "parent_task_ids_all": ",".join(super_task_ids) if len(super_task_ids) > 1 else None,
                "parent_folder_ids": ",".join(parent_folder_ids) if parent_folder_ids else None,
                "child_task_ids": ",".join(sub_task_ids) if sub_task_ids else None,
                "child_task_count": len(sub_task_ids),
                "current_status_id": status_id,
                "current_status_name": status_names.get(str(status_id or ""), status_id),
                "workflow_id": workflow_info.get("workflow_id"),
                "workflow_name": workflow_info.get("workflow_name"),
                "general_status": task.get("status"),
                "created_date": task.get("createdDate"),
                "updated_date": task.get("updatedDate"),
                "completed_date": task.get("completedDate"),
                "planned_start_date": dates.get("start"),
                "planned_due_date": dates.get("due"),
                "assignee_ids": ",".join(responsible_ids) if responsible_ids else None,
                "assignee_names": (
                    ",".join(contact_names.get(i, i) for i in responsible_ids)
                    if responsible_ids
                    else None
                ),
                "creator_id": author_ids[0] if author_ids else None,
                "creator_name": contact_names.get(author_ids[0]) if author_ids else None,
                "permalink": task.get("permalink"),
            }
        )

    frame = pd.DataFrame(rows, columns=TASK_INVENTORY_COLUMNS)
    if not frame.empty:
        frame = frame.sort_values(["item_type", "task_name"], na_position="last").reset_index(drop=True)
    return frame


# Wrike's own naming for the first six task-hierarchy levels, as seen in the
# actual Snowflake/Wrike export column headers (parent_task, child_task,
# grandchild_task, baby_task, grandbaby_task, great_grandbaby_task). A project
# nesting deeper than that is not something this codebase has seen evidence
# of, so levels past 6 fall back to a generic "level_N_task" name rather than
# guessing a Wrike-style word for them.
HIERARCHY_LEVEL_NAMES = [
    "parent_task",
    "child_task",
    "grandchild_task",
    "baby_task",
    "grandbaby_task",
    "great_grandbaby_task",
]


def _level_name(depth_index: int) -> str:
    if depth_index < len(HIERARCHY_LEVEL_NAMES):
        return HIERARCHY_LEVEL_NAMES[depth_index]
    return f"level_{depth_index + 1}_task"


def _ancestor_chain(task_id: str, tasks_by_id: Mapping[str, Mapping[str, Any]]) -> list[str]:
    """Walk superTaskIds upward from `task_id` to its root, returning [root, ..., task_id].

    Uses the first superTaskIds entry when a task lists more than one parent -
    Wrike allows multiple, but a single-parent tree is what every downstream
    column (parent_task/child_task/...) assumes. A `seen` guard stops any
    cyclic or self-referential superTaskIds data from looping forever.
    """
    chain: list[str] = []
    seen: set[str] = set()
    current: str | None = task_id
    while current is not None and current not in seen:
        seen.add(current)
        chain.append(current)
        task = tasks_by_id.get(current)
        supers = [str(i) for i in ((task.get("superTaskIds") or []) if task else [])]
        current = supers[0] if supers else None
    chain.reverse()
    return chain


def build_task_hierarchy_table(
    *,
    project_id: str,
    project_name: str | None,
    tasks_by_id: Mapping[str, Mapping[str, Any]],
    status_names: Mapping[str, str],
    contact_names: Mapping[str, str] | None = None,
    status_workflows: Mapping[str, Mapping[str, Any]] | None = None,
) -> pd.DataFrame:
    """One row per task/subtask, with its full ancestor path spelled out in
    named columns (parent_task_id/title, child_task_id/title, ...) instead of
    a single parent-pointer column - matching the shape of the real Wrike/
    Snowflake export. Column width is DYNAMIC: only as many level pairs as
    the deepest chain in `tasks_by_id` actually needs are emitted, so a
    project that never nests past "child_task" gets no baby_task/
    grandbaby_task columns at all, while one that reaches six levels gets all
    six. Every value traces back to Wrike's own superTaskIds/subTaskIds/
    customStatusId/etc. - nothing here is entered manually.

    `task_name` is carried as its own column (not just implicit in whichever
    ..._title column matches this row's own depth) so a reader doesn't have
    to cross-reference hierarchy_depth to find out what the task is called.
    `status_workflows` maps customStatusId -> {"workflow_id", "workflow_name"}
    (build via wrike_status_api.flatten_workflows on GET /workflows) so the
    workflow behind a task's status is visible, not just the status name.
    """
    contact_names = contact_names or {}
    status_workflows = status_workflows or {}
    chains = {task_id: _ancestor_chain(task_id, tasks_by_id) for task_id in tasks_by_id}
    max_depth = max((len(chain) for chain in chains.values()), default=0)
    level_names = [_level_name(i) for i in range(max_depth)]

    columns = ["project_id", "project_name", "task_id", "task_name"]
    for name in level_names:
        columns += [f"{name}_id", f"{name}_title"]
    columns += [
        "hierarchy_depth",
        "hierarchy_level",
        "current_status_id",
        "current_status_name",
        "workflow_id",
        "workflow_name",
        "general_status",
        "created_date",
        "updated_date",
        "completed_date",
        "planned_start_date",
        "planned_due_date",
        "assignee_ids",
        "assignee_names",
        "creator_id",
        "creator_name",
        "permalink",
    ]

    rows: list[dict[str, Any]] = []
    for task_id, task in tasks_by_id.items():
        chain = chains[task_id]
        row: dict[str, Any] = {
            "project_id": project_id,
            "project_name": project_name,
            "task_id": str(task_id),
            "task_name": task.get("title"),
        }
        for depth_index, name in enumerate(level_names):
            if depth_index < len(chain):
                ancestor_id = chain[depth_index]
                ancestor = tasks_by_id.get(ancestor_id, {})
                row[f"{name}_id"] = ancestor_id
                row[f"{name}_title"] = ancestor.get("title")
            else:
                row[f"{name}_id"] = None
                row[f"{name}_title"] = None

        responsible_ids = [str(i) for i in (task.get("responsibleIds") or [])]
        author_ids = [str(i) for i in (task.get("authorIds") or [])]
        dates = task.get("dates") if isinstance(task.get("dates"), Mapping) else {}
        status_id = task.get("customStatusId")
        workflow_info = status_workflows.get(str(status_id or ""), {})

        row.update(
            {
                "hierarchy_depth": len(chain),
                "hierarchy_level": _level_name(len(chain) - 1) if chain else None,
                "current_status_id": status_id,
                "current_status_name": status_names.get(str(status_id or ""), status_id),
                "workflow_id": workflow_info.get("workflow_id"),
                "workflow_name": workflow_info.get("workflow_name"),
                "general_status": task.get("status"),
                "created_date": task.get("createdDate"),
                "updated_date": task.get("updatedDate"),
                "completed_date": task.get("completedDate"),
                "planned_start_date": dates.get("start"),
                "planned_due_date": dates.get("due"),
                "assignee_ids": ",".join(responsible_ids) if responsible_ids else None,
                "assignee_names": (
                    ",".join(contact_names.get(i, i) for i in responsible_ids)
                    if responsible_ids
                    else None
                ),
                "creator_id": author_ids[0] if author_ids else None,
                "creator_name": contact_names.get(author_ids[0]) if author_ids else None,
                "permalink": task.get("permalink"),
            }
        )
        rows.append(row)

    frame = pd.DataFrame(rows, columns=columns)
    if not frame.empty:
        frame = frame.sort_values(["hierarchy_depth", "task_id"]).reset_index(drop=True)
    return frame


TASK_HISTORY_CLASSIFICATIONS = {
    "CURRENT_STATUS_ONLY",
    "HISTORY_AVAILABLE",
    "HISTORY_NOT_AVAILABLE",
    "INSUFFICIENT_EVIDENCE",
    "TASK_NOT_FOUND",
}

# Fields that would indicate Wrike returned an actual status-change record on
# the Task object itself, as opposed to just its current state. Checked
# defensively rather than assumed absent - if a future Wrike response ever
# includes one of these, this function reports HISTORY_AVAILABLE instead of
# silently missing it.
_TASK_HISTORY_SHAPED_FIELDS = ("statusHistory", "history", "transitions", "auditTrail", "statusChanges")


def classify_task_history_availability(
    *,
    tasks_by_id: Mapping[str, Mapping[str, Any]],
    status_names: Mapping[str, str] | None = None,
    expected_task_ids: Iterable[str] | None = None,
    audit_log_blocked_reason: str | None = None,
) -> pd.DataFrame:
    """Classify, per task, exactly what the accessible Wrike sources can prove
    about its status history - this is Akash's assignment step 6, not a
    blanket assertion.

    For each task this checks, in order: is it missing entirely
    (TASK_NOT_FOUND); does its Task object carry any history-shaped field
    (HISTORY_AVAILABLE - checked, not assumed absent); does it even have a
    current status to report (INSUFFICIENT_EVIDENCE if not); and only then
    does it land on CURRENT_STATUS_ONLY, and only with `audit_log_blocked_reason`
    explaining why Audit Log - the one documented source of TaskStatusChanged
    events - couldn't be used to fill the gap either. Pass `expected_task_ids`
    (e.g. from a Snowflake/Excel export) to also catch tasks that should exist
    but were never returned by the API pull at all.
    """
    status_names = status_names or {}
    expected_ids = sorted({str(i) for i in (expected_task_ids or tasks_by_id.keys())})
    rows: list[dict[str, Any]] = []
    for task_id in expected_ids:
        task = tasks_by_id.get(task_id)
        if task is None:
            rows.append(
                {
                    "task_id": task_id,
                    "task_name": None,
                    "current_status_id": None,
                    "current_status_name": None,
                    "classification": "TASK_NOT_FOUND",
                    "evidence": (
                        "Not returned by GET /folders/{id}/tasks nor found while "
                        "closing over subTaskIds - this task/subtask could not be "
                        "retrieved at all with the current token/access."
                    ),
                }
            )
            continue

        history_fields_present = sorted(f for f in _TASK_HISTORY_SHAPED_FIELDS if task.get(f))
        status_id = task.get("customStatusId")

        if history_fields_present:
            classification = "HISTORY_AVAILABLE"
            evidence = (
                f"Task object unexpectedly carries {history_fields_present} - "
                "inspect this field directly, it may contain real transition data."
            )
        elif status_id is None:
            classification = "INSUFFICIENT_EVIDENCE"
            evidence = "Task object has no customStatusId at all - cannot even confirm its current state."
        elif audit_log_blocked_reason:
            classification = "CURRENT_STATUS_ONLY"
            evidence = (
                "Task object exposes only customStatusId (current state) - no "
                "prior-status field, transition timestamp, or ordered history "
                "array is present. Audit Log, the only Wrike endpoint documented "
                "to return TaskStatusChanged events, is blocked for this token: "
                f"{audit_log_blocked_reason}"
            )
        else:
            classification = "INSUFFICIENT_EVIDENCE"
            evidence = (
                "Task object exposes only customStatusId; Audit Log was not "
                "reported as blocked for this run, but was not actually queried "
                "for this task either - re-run with a live Audit Log attempt "
                "before treating this as CURRENT_STATUS_ONLY."
            )

        rows.append(
            {
                "task_id": task_id,
                "task_name": task.get("title"),
                "current_status_id": status_id,
                "current_status_name": status_names.get(str(status_id or ""), status_id),
                "classification": classification,
                "evidence": evidence,
            }
        )

    return pd.DataFrame(
        rows,
        columns=["task_id", "task_name", "current_status_id", "current_status_name", "classification", "evidence"],
    )


def reconcile_task_inventory(
    inventory: pd.DataFrame, expected: pd.DataFrame, *, expected_id_col: str = "task_id"
) -> dict[str, Any]:
    """Compare a pulled task_inventory against an expected list (Excel/Snowflake export).

    Pure set comparison on task_id - the counts this returns are exactly the
    ones Akash's assignment asks for (expected/returned/missing/duplicate/
    match percentage). Duplicates are detected in `inventory` itself, since a
    correct pull should have exactly one row per task_id.
    """
    expected_ids = set(expected[expected_id_col].astype(str))
    returned_ids = set(inventory["task_id"].astype(str)) if not inventory.empty else set()
    duplicate_ids = (
        inventory["task_id"].astype(str).value_counts().loc[lambda s: s > 1].index.tolist()
        if not inventory.empty
        else []
    )
    missing_ids = sorted(expected_ids - returned_ids)
    unexpected_ids = sorted(returned_ids - expected_ids)

    expected_tasks = int((expected.get("item_type", pd.Series(dtype=str)).astype(str).str.upper() == "TASK").sum()) if "item_type" in expected.columns else None
    expected_subtasks = int((expected.get("item_type", pd.Series(dtype=str)).astype(str).str.upper() == "SUBTASK").sum()) if "item_type" in expected.columns else None
    returned_tasks = int((inventory["item_type"] == "TASK").sum()) if not inventory.empty else 0
    returned_subtasks = int((inventory["item_type"] == "SUBTASK").sum()) if not inventory.empty else 0

    match_percentage = (
        round(100 * len(expected_ids & returned_ids) / len(expected_ids), 2) if expected_ids else None
    )

    return {
        "expected_total": len(expected_ids),
        "expected_tasks": expected_tasks,
        "expected_subtasks": expected_subtasks,
        "returned_total": len(returned_ids),
        "returned_tasks": returned_tasks,
        "returned_subtasks": returned_subtasks,
        "missing_count": len(missing_ids),
        "missing_ids": missing_ids,
        "duplicate_count": len(duplicate_ids),
        "duplicate_ids": duplicate_ids,
        "unexpected_count": len(unexpected_ids),
        "unexpected_ids": unexpected_ids,
        "match_percentage": match_percentage,
    }


def verify_control_project(
    project: Mapping[str, Any],
    *,
    status_names: Mapping[str, str] | None = None,
    history: pd.DataFrame | None = None,
    expectations: Iterable[Mapping[str, str]] = CONTROL_EXPECTATIONS,
    access_checks: Iterable[Mapping[str, Any]] | None = None,
) -> tuple[dict[str, Any], pd.DataFrame]:
    """Compare Akash's control-project expectations with real evidence.

    The ordinary project response is current-state metadata, not a status
    timeline. It can independently confirm the Completed transition through
    ``completedDate`` but it cannot prove when the project became In Progress.
    Only Audit Log or BI Export rows count as historical-backfill evidence.

    Synthetic sample rows are deliberately rejected even if their status/date
    values happen to match an expectation.
    """
    status_names = status_names or {}
    expected = [dict(item) for item in expectations]
    project_id = str(project.get("id") or "")
    project_name = project.get("title")
    nested_project = (
        project.get("project")
        if isinstance(project.get("project"), Mapping)
        else {}
    )
    current_status_id = project.get("customStatusId") or nested_project.get(
        "customStatusId"
    )
    current_status_name = (
        project.get("status")
        or nested_project.get("status")
        or status_names.get(str(current_status_id or ""))
    )
    completed_at = _utc(project.get("completedDate") or nested_project.get("completedDate"))

    candidate_history = (
        finalize_history(history.copy())
        if history is not None
        else finalize_history(pd.DataFrame())
    )
    if not candidate_history.empty:
        candidate_history = candidate_history[
            candidate_history["project_id"].astype(str).eq(project_id)
        ].copy()

    rows: list[dict[str, Any]] = []
    for expectation in expected:
        wanted_status = str(expectation["status_name"])
        wanted_date = pd.Timestamp(expectation["status_changed_date"]).date()
        matches = candidate_history.iloc[0:0].copy()
        if not candidate_history.empty:
            normalized_names = (
                candidate_history["status_name"]
                .fillna("")
                .astype(str)
                .str.strip()
                .str.casefold()
            )
            changed_dates = pd.to_datetime(
                candidate_history["status_changed_at"], utc=True, errors="coerce"
            ).dt.date
            matches = candidate_history[
                normalized_names.eq(wanted_status.strip().casefold())
                & changed_dates.eq(wanted_date)
            ].copy()

        rejected_synthetic = False
        live_match: Mapping[str, Any] | None = None
        if not matches.empty:
            sources = matches["source"].fillna("").astype(str).str.upper()
            event_ids = matches["source_event_id"].fillna("").astype(str).str.lower()
            synthetic_mask = sources.str.contains(
                "SYNTHETIC", regex=False
            ) | event_ids.str.startswith("demo-event-")
            rejected_synthetic = bool(synthetic_mask.any())
            live_matches = matches[
                sources.isin(HISTORICAL_BACKFILL_SOURCES) & ~synthetic_mask
            ]
            if not live_matches.empty:
                live_match = live_matches.sort_values("status_changed_at").iloc[0].to_dict()

        if live_match is not None:
            result = "CONFIRMED_LIVE_HISTORY"
            source = str(live_match.get("source") or "")
            observed_at = pd.to_datetime(
                live_match.get("status_changed_at"), utc=True, errors="coerce"
            )
            explanation = "Matched a historical event returned by Audit Log or BI Export."
        elif (
            wanted_status.strip().casefold() == "completed"
            and str(current_status_name or "").strip().casefold() == "completed"
            and not pd.isna(completed_at)
            and completed_at.date() == wanted_date
        ):
            result = "CONFIRMED_CURRENT_METADATA_ONLY"
            source = "CURRENT_PROJECT_METADATA"
            observed_at = completed_at
            explanation = (
                "completedDate confirms this transition, but the project response "
                "still is not a complete status timeline."
            )
        elif rejected_synthetic:
            result = "REJECTED_SYNTHETIC_MATCH"
            source = "SYNTHETIC_DEMO"
            observed_at = pd.NaT
            explanation = "A matching demo row exists, but fabricated events are not evidence."
        else:
            result = "MISSING_FROM_ACCESSIBLE_SOURCES"
            source = None
            observed_at = pd.NaT
            explanation = (
                "The transition is absent from current project metadata and no "
                "permitted historical-backfill row was supplied."
            )

        rows.append(
            {
                "project_id": project_id,
                "project_name": project_name,
                "expected_status": wanted_status,
                "expected_date": wanted_date.isoformat(),
                "result": result,
                "evidence_source": source,
                "observed_at": None if pd.isna(observed_at) else observed_at.isoformat(),
                "explanation": explanation,
            }
        )

    checks = [dict(check) for check in (access_checks or [])]
    blocked_historical_paths = [
        check
        for check in checks
        if str(check.get("purpose") or "").startswith("HISTORICAL")
        and not bool(check.get("accessible"))
    ]

    if not expected:
        # No expectations were supplied for this project. Silently reusing
        # CONTROL_EXPECTATIONS (Akash's May 26 / June 10 facts, which apply
        # ONLY to the control project) would produce a result that looks
        # like a real check but isn't - it tests dates nobody claimed for
        # this project. Report that plainly instead of pretending to verify.
        acceptance_status = "NO_EXPECTATIONS_PROVIDED"
        all_live = False
        missing: list[dict[str, Any]] = []
        conclusion = (
            "No expected status/date pairs were supplied for this project, so "
            "nothing was verified. Only current-state facts are available: "
            f"current status is {current_status_name!r}"
            + (f", completed at {completed_at.isoformat()}" if not pd.isna(completed_at) else "")
            + ". Pass --expected to check specific transitions for this project."
        )
    else:
        all_live = bool(rows) and all(row["result"] == "CONFIRMED_LIVE_HISTORY" for row in rows)
        missing = [row for row in rows if row["result"] != "CONFIRMED_LIVE_HISTORY"]
        acceptance_status = "PASS_LIVE_HISTORY" if all_live else "BLOCKED_OR_INCOMPLETE"
        if all_live:
            conclusion = "All expected project transitions were confirmed from a historical source."
        else:
            # A row can be CONFIRMED_LIVE_HISTORY, CONFIRMED_CURRENT_METADATA_ONLY
            # (partial - real but not a historical record), REJECTED_SYNTHETIC_MATCH,
            # or MISSING_FROM_ACCESSIBLE_SOURCES (nothing at all). Collapsing all
            # non-live results into one "could not be confirmed" bucket erased that
            # distinction - restoring it so "we have SOME evidence" doesn't read the
            # same as "we have NONE".
            def describe(group: list[dict[str, Any]]) -> str:
                return "; ".join(f"{r['expected_status']} ({r['expected_date']})" for r in group)

            metadata_only = [r for r in missing if r["result"] == "CONFIRMED_CURRENT_METADATA_ONLY"]
            truly_missing = [r for r in missing if r["result"] == "MISSING_FROM_ACCESSIBLE_SOURCES"]
            rejected = [r for r in missing if r["result"] == "REJECTED_SYNTHETIC_MATCH"]

            parts = []
            if metadata_only:
                parts.append(
                    "confirmed only from current project metadata, not a historical record: "
                    + describe(metadata_only)
                )
            if truly_missing:
                parts.append("absent from every accessible source: " + describe(truly_missing))
            if rejected:
                parts.append(
                    "only matched by a rejected synthetic/demo row: " + describe(rejected)
                )

            live_count = len(rows) - len(missing)
            conclusion = (
                f"{live_count} of {len(rows)} expected transition(s) confirmed from live "
                f"historical evidence. {' | '.join(parts)}. Do not present synthetic rows or "
                "the workflow catalog as project history."
            )

    report = {
        "control_project_id": project_id,
        "control_project_name": project_name,
        "current_status_id": current_status_id,
        "current_status_name": current_status_name,
        "completed_at": None if pd.isna(completed_at) else completed_at.isoformat(),
        "acceptance_status": acceptance_status,
        "can_answer_akash_from_live_history": all_live,
        "expected_event_count": len(rows),
        "live_history_event_count": sum(
            row["result"] == "CONFIRMED_LIVE_HISTORY" for row in rows
        ),
        "events_not_confirmed_from_live_history": len(missing),
        "blocked_historical_paths": len(blocked_historical_paths),
        "event_checks": rows,
        "access_checks": checks,
        "conclusion": conclusion,
    }
    return report, pd.DataFrame(rows)


def _find_file(directory: Path, stem: str) -> Path | None:
    stem = stem.lower()
    for path in directory.rglob("*"):
        if path.is_file() and path.suffix.lower() in {".csv", ".tsv"} and path.stem.lower() == stem:
            return path
    return None


def _read_delimited(path: Path) -> pd.DataFrame:
    return pd.read_csv(path, sep="\t" if path.suffix.lower() == ".tsv" else ",", low_memory=False)


def _col(frame: pd.DataFrame, *candidates: str, required: bool = True) -> str | None:
    lookup = {str(c).lower(): str(c) for c in frame.columns}
    for candidate in candidates:
        if candidate.lower() in lookup:
            return lookup[candidate.lower()]
    if required:
        raise ValueError(f"Expected one of {candidates}; available columns: {list(frame.columns)}")
    return None


def normalize_bi_export(directory: str | Path) -> tuple[pd.DataFrame, dict[str, Any]]:
    """Normalize BI Export status history if project rows are present.

    Wrike documents ``work_workflow_stage_history`` as task-status history. The
    function therefore filters empirically for Project work items and reports a
    clear diagnostic instead of claiming project history is available.
    """
    root = Path(directory)
    diagnostics: dict[str, Any] = {"source": "BI_EXPORT", "directory": str(root)}
    item_path = _find_file(root, "work_item")
    history_path = _find_file(root, "work_workflow_stage_history")
    stage_path = _find_file(root, "workflow_stage")
    revision_path = _find_file(root, "work_change_revision")
    missing_files = [
        name
        for name, path in {
            "work_item": item_path,
            "work_workflow_stage_history": history_path,
            "workflow_stage": stage_path,
        }.items()
        if path is None
    ]
    if missing_files:
        diagnostics.update(status="blocked", reason=f"Missing BI files: {missing_files}")
        return pd.DataFrame(columns=HISTORY_COLUMNS), diagnostics

    items = _read_delimited(item_path)  # type: ignore[arg-type]
    history = _read_delimited(history_path)  # type: ignore[arg-type]
    stages = _read_delimited(stage_path)  # type: ignore[arg-type]
    item_id = _col(items, "work_id", "id")
    item_type = _col(items, "work_type", "type")
    item_api_id = _col(items, "api_id", "work_api_id", "work_item_api_id", required=False)
    item_name = _col(items, "work_name", "name", "title", required=False)
    hist_work_id = _col(history, "work_id")
    new_stage = _col(history, "new_workflow_stage_id", "workflow_stage_id", "new_status_id")
    old_stage = _col(history, "old_workflow_stage_id", "old_status_id", required=False)
    author = _col(history, "user_id", "author_id", required=False)
    event_id = _col(history, "work_workflow_stage_history_id", "id", required=False)
    changed_at = _col(history, "when_work_changed", "changed_at", "created_date", required=False)

    merged = history.merge(
        items[[c for c in [item_id, item_type, item_api_id, item_name] if c]],
        left_on=hist_work_id,
        right_on=item_id,
        how="left",
        suffixes=("", "_item"),
    )
    merged = merged[merged[item_type].astype(str).str.lower().eq("project")].copy()
    diagnostics["project_status_rows_found"] = int(len(merged))
    if merged.empty:
        diagnostics.update(
            status="unsupported_or_empty",
            reason="No Project rows exist in work_workflow_stage_history; Wrike documents this table as task status history.",
        )
        return pd.DataFrame(columns=HISTORY_COLUMNS), diagnostics

    if changed_at is None and revision_path is not None:
        revisions = _read_delimited(revision_path)
        revision_id_hist = _col(history, "work_change_revision_id", "revision_id", required=False)
        revision_id = _col(revisions, "work_change_revision_id", "id", "revision_id", required=False)
        revision_time = _col(revisions, "when_work_changed", "changed_at", required=False)
        if revision_id_hist and revision_id and revision_time:
            merged = merged.merge(
                revisions[[revision_id, revision_time]],
                left_on=revision_id_hist,
                right_on=revision_id,
                how="left",
            )
            changed_at = revision_time
    if changed_at is None:
        diagnostics.update(status="blocked", reason="Project history rows have no resolvable change timestamp.")
        return pd.DataFrame(columns=HISTORY_COLUMNS), diagnostics

    stage_id = _col(stages, "workflow_stage_id", "id", "status_id")
    stage_name = _col(stages, "workflow_stage_name", "name", "status_name")
    status_map = stages.set_index(stage_id)[stage_name].astype(str).to_dict()
    rows: list[dict[str, Any]] = []
    for _, raw in merged.iterrows():
        pid = raw.get(item_api_id) if item_api_id else raw.get(item_id)
        new_id = raw.get(new_stage)
        old_id = raw.get(old_stage) if old_stage else None
        row = {
            "project_id": str(pid),
            "project_name": raw.get(item_name) if item_name else None,
            "old_status_id": old_id,
            "old_status_name": status_map.get(old_id),
            "status_id": new_id,
            "status_name": status_map.get(new_id),
            "status_changed_at": _utc(raw.get(changed_at)),
            "event_author_id": raw.get(author) if author else None,
            "source": "BI_EXPORT",
            "source_event_id": str(raw.get(event_id)) if event_id and pd.notna(raw.get(event_id)) else None,
            "source_payload": json.dumps(raw.dropna().to_dict(), separators=(",", ":"), default=str),
            "ingested_at": pd.Timestamp.now(tz="UTC"),
        }
        row["source_event_id"] = row["source_event_id"] or _stable_event_id(row)
        rows.append(row)
    output = finalize_history(pd.DataFrame(rows))
    diagnostics.update(status="ok", normalized_rows=int(len(output)))
    return output, diagnostics


def lifecycle_metrics(history: pd.DataFrame) -> pd.DataFrame:
    """Return time spent between consecutive project statuses in hours/days."""
    columns = [
        "project_id", "project_name", "from_status", "to_status",
        "from_at", "to_at", "duration_hours", "duration_days", "transition_sequence",
    ]
    if history.empty:
        return pd.DataFrame(columns=columns)
    rows: list[dict[str, Any]] = []
    for project_id, events in history.groupby("project_id", sort=False):
        ordered = events.dropna(subset=["status_changed_at"]).sort_values("status_changed_at")
        for index in range(1, len(ordered)):
            previous = ordered.iloc[index - 1]
            current = ordered.iloc[index]
            delta = current["status_changed_at"] - previous["status_changed_at"]
            rows.append({
                "project_id": project_id,
                "project_name": current.get("project_name") or previous.get("project_name"),
                "from_status": previous.get("status_name"),
                "to_status": current.get("status_name"),
                "from_at": previous["status_changed_at"],
                "to_at": current["status_changed_at"],
                "duration_hours": delta.total_seconds() / 3600,
                "duration_days": delta.total_seconds() / 86400,
                "transition_sequence": index,
            })
    return pd.DataFrame(rows, columns=columns)


def history_json_map(history: pd.DataFrame) -> dict[str, str]:
    output: dict[str, str] = {}
    for project_id, events in history.groupby("project_id", sort=False):
        records = []
        for row in events.sort_values("event_sequence").to_dict("records"):
            changed_at = row.get("status_changed_at")
            records.append({
                "sequence": int(row["event_sequence"]),
                "old_status_id": row.get("old_status_id"),
                "old_status_name": row.get("old_status_name"),
                "status_id": row.get("status_id"),
                "status_name": row.get("status_name"),
                "changed_at": None if pd.isna(changed_at) else pd.Timestamp(changed_at).isoformat().replace("+00:00", "Z"),
                "event_author_id": row.get("event_author_id"),
                "source": row.get("source"),
            })
        output[str(project_id)] = json.dumps(records, separators=(",", ":"), default=str)
    return output


def attach_to_project_rows(
    wrike_stg: pd.DataFrame,
    history: pd.DataFrame,
    *,
    project_key_col: str = "key",
    is_project_col: str = "is_project",
) -> pd.DataFrame:
    """Add history fields only to ``is_project=Y`` rows, preserving row count."""
    missing = {project_key_col, is_project_col}.difference(wrike_stg.columns)
    if missing:
        raise KeyError(f"Missing Wrike_STG columns: {sorted(missing)}")
    before = len(wrike_stg)
    result = wrike_stg.copy()
    added = [
        "PROJECT_STATUS_HISTORY_JSON", "PROJECT_LATEST_STATUS_ID",
        "PROJECT_LATEST_STATUS", "PROJECT_LATEST_STATUS_CHANGED_AT",
        "PROJECT_STATUS_EVENT_COUNT",
    ]
    result["PROJECT_STATUS_HISTORY_JSON"] = pd.NA
    result["PROJECT_LATEST_STATUS_ID"] = pd.NA
    result["PROJECT_LATEST_STATUS"] = pd.NA
    result["PROJECT_LATEST_STATUS_CHANGED_AT"] = pd.Series(
        pd.NaT, index=result.index, dtype="datetime64[ns, UTC]"
    )
    result["PROJECT_STATUS_EVENT_COUNT"] = pd.NA
    project_mask = result[is_project_col].astype(str).str.upper().eq("Y")
    if not history.empty:
        latest = history.sort_values(["project_id", "event_sequence"]).groupby("project_id").tail(1).set_index("project_id")
        keys = result.loc[project_mask, project_key_col].astype(str)
        result.loc[project_mask, "PROJECT_STATUS_HISTORY_JSON"] = keys.map(history_json_map(history)).values
        result.loc[project_mask, "PROJECT_LATEST_STATUS_ID"] = keys.map(latest["status_id"].to_dict()).values
        result.loc[project_mask, "PROJECT_LATEST_STATUS"] = keys.map(latest["status_name"].to_dict()).values
        result.loc[project_mask, "PROJECT_LATEST_STATUS_CHANGED_AT"] = keys.map(
            latest["status_changed_at"].to_dict()
        )
        result.loc[project_mask, "PROJECT_STATUS_EVENT_COUNT"] = keys.map(history.groupby("project_id").size().to_dict()).values
    if len(result) != before:
        raise AssertionError("Wrike_STG row count changed")
    if result.loc[~project_mask, added].notna().any().any():
        raise AssertionError("Project history leaked onto task/subtask rows")
    return result
