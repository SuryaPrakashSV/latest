WITH task_rows AS (
    SELECT
        "id" AS project_id,

        COALESCE(
            CASE WHEN TRIM("great_grandbaby_task_id") <> ''
                 THEN "great_grandbaby_task_id" END,
            CASE WHEN TRIM("grandbaby_task_id") <> ''
                 THEN "grandbaby_task_id" END,
            CASE WHEN TRIM("baby_task_id") <> ''
                 THEN "baby_task_id" END,
            CASE WHEN TRIM("grandchild_task_id") <> ''
                 THEN "grandchild_task_id" END,
            CASE WHEN TRIM("child_task_id") <> ''
                 THEN "child_task_id" END,
            CASE WHEN TRIM("parent_task_id") <> ''
                 THEN "parent_task_id" END
        ) AS task_id,

        "status" AS general_status,
        "customStatusId" AS detailed_status,
        "updatedDate" AS updated_date,
        "completedDate" AS completed_date

    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
)
SELECT
    task_id,
    project_id,
    general_status,
    detailed_status,
    updated_date,
    completed_date,
    COUNT(*) AS physical_row_count
FROM task_rows
WHERE task_id IN (
    'IEADQP2VKRD2ARI6',  -- Parent-level task
    'IEADQP2VKRD2ARJD',  -- Parent-level task
    'IEADQP2VKRD2NCRG',  -- Parent-level task
    'IEADQP2VKRD3J35Z',  -- Child task
    'IEADQP2VKRD3J37H',  -- Grandchild task
    'IEADQP2VKRD3J37N'   -- Grandchild task
)
GROUP BY
    task_id,
    project_id,
    general_status,
    detailed_status,
    updated_date,
    completed_date
ORDER BY
    task_id,
    updated_date,
    project_id,
    general_status,
    detailed_status,
    completed_date;
