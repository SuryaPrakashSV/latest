WITH project_tasks AS (
    SELECT
        "id" AS project_id,
        "title" AS project_name,
        COALESCE(
            NULLIF(TRIM("great_grandbaby_task_id"), ''),
            NULLIF(TRIM("grandbaby_task_id"), ''),
            NULLIF(TRIM("baby_task_id"), ''),
            NULLIF(TRIM("grandchild_task_id"), ''),
            NULLIF(TRIM("child_task_id"), ''),
            NULLIF(TRIM("parent_task_id"), '')
        ) AS task_id,
        "status",
        "customStatusId",
        "updatedDate"
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
    WHERE "title" IN (
        'Apple Shopping Event (FY27)',
        'Holiday (FY27)'
    )
)
SELECT
    project_id,
    project_name,
    task_id,
    "status",
    "customStatusId",
    MAX("updatedDate") AS latest_updated_date,
    COUNT(*) AS row_count
FROM project_tasks
WHERE task_id IS NOT NULL
GROUP BY
    project_id,
    project_name,
    task_id,
    "status",
    "customStatusId"
ORDER BY project_name, latest_updated_date DESC;
