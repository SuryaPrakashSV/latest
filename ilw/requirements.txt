WITH project_rows AS (
    SELECT
        "key"            AS entity_id,
        "title"          AS title,
        "status"         AS general_status,
        "customStatusId" AS detailed_status,
        "updatedDate"    AS updated_date,
        CASE
            WHEN "key" IN ("parent_task_id","child_task_id","grandchild_task_id",
                           "baby_task_id","grandbaby_task_id","great_grandbaby_task_id")
            THEN 'task'
            ELSE 'project / other'
        END AS entity_kind
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
    WHERE "id" = '<PASTE_PROJECT_ID_HERE>'
      AND NULLIF(TRIM("key"), '') IS NOT NULL
      AND UPPER(TRIM("key")) <> 'PLACEHOLDER'
    QUALIFY ROW_NUMBER() OVER (
        PARTITION BY "key"
        ORDER BY TRY_TO_TIMESTAMP_TZ("updatedDate"::VARCHAR) DESC NULLS LAST, "id"
    ) = 1
)
SELECT
    entity_kind,
    entity_id,
    title,
    general_status,
    detailed_status,
    updated_date
FROM project_rows
ORDER BY entity_kind, detailed_status, title;
