WITH latest_tasks AS (
    SELECT
        "id"             AS project_api_id,
        "key"            AS task_id,
        "customStatusId" AS detailed_status
    FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
    WHERE NULLIF(TRIM("key"), '') IS NOT NULL
      AND UPPER(TRIM("key")) <> 'PLACEHOLDER'
      AND "key" IN ("parent_task_id","child_task_id","grandchild_task_id",
                    "baby_task_id","grandbaby_task_id","great_grandbaby_task_id")
    QUALIFY ROW_NUMBER() OVER (
        PARTITION BY "key"
        ORDER BY TRY_TO_TIMESTAMP_TZ("updatedDate"::VARCHAR) DESC NULLS LAST, "id"
    ) = 1
)
SELECT
    project_api_id,
    COUNT(*) AS active_task_count
FROM latest_tasks
WHERE LOWER(TRIM(detailed_status)) IN ('in progress','new','not started','t - in progress')
GROUP BY project_api_id
ORDER BY active_task_count DESC
LIMIT 20;
