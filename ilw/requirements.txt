SELECT
    "id" AS project_id,
    "title" AS project_name,
    SUBSTR("id", 8, 1) AS ambiguous_char,
    ASCII(SUBSTR("id", 8, 1)) AS ambiguous_char_code
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
WHERE COALESCE(TRY_TO_BOOLEAN("is_project"), FALSE) = TRUE
  AND "title" ILIKE 'Holiday (FY27)%'
GROUP BY "id", "title";
