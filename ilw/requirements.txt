SELECT
    COLUMN_NAME,
    DATA_TYPE
FROM GBI_RETAIL_BAP_DB.INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'ASO_OPS_WSA_SNDBX_BIZ_APP'
  AND TABLE_NAME = 'Wrike_STG'
  AND (
      COLUMN_NAME ILIKE '%effort%'
      OR COLUMN_NAME ILIKE '%hour%'
      OR COLUMN_NAME ILIKE '%allocat%'
      OR COLUMN_NAME ILIKE '%estimat%'
      OR COLUMN_NAME ILIKE '%duration%'
      OR COLUMN_NAME ILIKE '%timelog%'
      OR COLUMN_NAME ILIKE '%time_spent%'
  )
ORDER BY ORDINAL_POSITION;

WITH targets AS (
    SELECT column1 AS container_name
    FROM VALUES
        ('Test Import'),
        ('Intake Form - Resourcing Template'),
        ('NPI Project Blueprints'),
        ('xxOLD Blueprints')
),
containers AS (
    SELECT DISTINCT
        t.container_name,
        w."id" AS container_id
    FROM targets t
    LEFT JOIN
        GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG" w
        ON TRIM(w."title") = t.container_name
       AND w."key" = w."id"
)
SELECT DISTINCT
    c.container_name,
    c.container_id,
    w."key"   AS entity_id,
    w."title" AS entity_title,

    CASE
        WHEN c.container_id IS NULL THEN 'CONTAINER_NOT_FOUND'
        WHEN w."key" = w."id" THEN 'CONTAINER_OWN_ROW'
        ELSE 'ASSOCIATED_ROW'
    END AS row_kind,

    w."effortAllocation_mode",
    w."effortAllocation_totalEffort",
    w."effortAllocation_allocatedEffort",
    w."totalEffortMinutes",
    w."owner_effortMinutes_per_day",

    w."effort [days]",
    w."total hours",
    w."allocated hours",
    w."booked effort",
    w."forecasted hours",
    w."[xa] effort h/d",
    w."[xa] effort total p/d",
    w."[xa] mission control effort"

FROM containers c
LEFT JOIN
    GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG" w
    ON w."id" = c.container_id

ORDER BY
    container_name,
    entity_id;







SELECT
    COLUMN_NAME,
    DATA_TYPE
FROM GBI_RETAIL_BAP_DB.INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'ASO_OPS_WSA_SNDBX_BIZ_APP'
  AND TABLE_NAME = 'Wrike_STG'
  AND (
      COLUMN_NAME ILIKE '%folder%'
      OR COLUMN_NAME ILIKE '%parentid%'
      OR COLUMN_NAME ILIKE '%parent_id%'
  )
ORDER BY ORDINAL_POSITION;






WITH targets AS (
    SELECT column1 AS container_name
    FROM VALUES
        ('Test Import'),
        ('Intake Form - Resourcing Template'),
        ('NPI Project Blueprints'),
        ('xxOLD Blueprints')
),
containers AS (
    SELECT DISTINCT
        t.container_name,
        w."id" AS container_id
    FROM targets t
    LEFT JOIN
        GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG" w
        ON TRIM(w."title") = t.container_name
       AND w."key" = w."id"
)
SELECT DISTINCT
    c.container_name,
    c.container_id,

    w."id"    AS associated_container_id,
    w."key"   AS entity_id,
    w."title" AS entity_title,

    CASE
        WHEN c.container_id IS NULL
            THEN 'CONTAINER_NOT_FOUND'
        WHEN w."key" = c.container_id
            THEN 'CONTAINER_OWN_ROW'
        WHEN w."id" = c.container_id
            THEN 'ASSOCIATED_BY_ID'
        ELSE 'MATCHED_FOLDER_HIERARCHY'
    END AS match_type,

    w."effortAllocation_mode",
    w."effortAllocation_totalEffort",
    w."effortAllocation_allocatedEffort",
    w."totalEffortMinutes",
    w."owner_effortMinutes_per_day",

    w."effort [days]",
    w."total hours",
    w."allocated hours",
    w."booked effort",
    w."forecasted hours",
    w."[xa] effort h/d",
    w."[xa] effort total p/d",
    w."[xa] mission control effort"

FROM containers c
LEFT JOIN
    GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG" w
    ON c.container_id IN (
        w."id",
        w."parent_folder_id",
        w."child_folder_id",
        w."grandchild_folder_id",
        w."baby_folder_id",
        w."grandbaby_folder_id"
    )

ORDER BY
    container_name,
    associated_container_id,
    entity_id;



SELECT *
FROM GBI_RETAIL_BAP_DB.ASO_OPS_WSA_SNDBX_BIZ_APP."Wrike_STG"
WHERE TRIM("key") = TRIM("id")
  AND TRIM("title") IN (
      'Test Import',
      'Intake Form - Resourcing Template',
      'NPI Project Blueprints',
      'xxOLD Blueprints'
  )
ORDER BY "title", "id";
