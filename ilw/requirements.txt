import ast
from pathlib import Path
from datetime import datetime

root = Path.cwd()
source = root / "Wrike_Data_local_validation.py"
tree = ast.parse(source.read_text(encoding="utf-8-sig"))

print("Remaining Snowflake calls:")
for node in ast.walk(tree):
    if isinstance(node, ast.Call):
        name = getattr(node.func, "id", getattr(node.func, "attr", ""))
        if name in {"get_snowflake_connection", "write_pandas"}:
            print(f"  Line {node.lineno}: {name}")

print("\nExisting CSV files:")
for path in sorted(root.rglob("*.csv")):
    info = path.stat()
    modified = datetime.fromtimestamp(info.st_mtime).isoformat(timespec="seconds")
    print(f"{modified} | {info.st_size:,} bytes | {path.relative_to(root)}")
PY
