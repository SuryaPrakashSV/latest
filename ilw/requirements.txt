# ==================== LOCAL CSV EXPORT ====================
from pathlib import Path
from datetime import datetime, timezone

if df_final_complete.empty:
    raise RuntimeError(
        "Extraction returned no rows. Local validation cannot proceed."
    )

run_id = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%S_%fZ")

output_dir = (
    Path(__file__).resolve().parent
    / "output"
    / "local_validation"
    / run_id
)
output_dir.mkdir(parents=True, exist_ok=False)

output_file = output_dir / "wrike_local_full.csv"

df_final_complete.to_csv(
    output_file,
    index=False,
    encoding="utf-8-sig",
)

print(f"\nLocal CSV exported: {output_file}")
print(f"Total rows exported: {len(df_final_complete):,}")
print("These are dataset rows, not a unique project count.")
